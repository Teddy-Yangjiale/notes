  44 def resolve_attention_backend(requested: str) -> str:
  45     """Resolve auto to paged when possible; explicit paged never falls back."""
  46     if requested == "sdpa":
  47         return "sdpa"
  48     available = flashinfer_available()
  49     if requested == "paged" and not available:
  50         raise RuntimeError("attention_backend='paged' requires a working FlashInfer installation")
  51     return "paged" if available else "sdpa"

  54 def disable_unavailable_packaged_flash_attention(model: torch.nn.Module) -> int:
  55     """Select SDPA once instead of entering exception-driven fallback per layer."""
  56     disabled = 0
  57     for module in model.modules():
  58         if not getattr(module, "use_packaged_flash_attn", False):
  59             continue
  60         owner = importlib.import_module(module.__class__.__module__)
  61         if getattr(owner, "_flash_attn_func_bhld_optional", None) is None:
  62             module.use_packaged_flash_attn = False
  63             disabled += 1
  64     return disabled

 124     def _frames(self, paths: Iterable[Path]):
 125         for tensor, _ in iter_preprocessed(
 126             paths, height=self.config.height, width=self.config.width
 127         ):
 128             observer = getattr(self, "_image_observer", None)
 129             if observer is not None:
 130                 # The loop descriptor worker consumes the exact CPU tensor that
 131                 # is sent to the streaming model, avoiding a second decode path.
 132                 observer(tensor.unsqueeze(0).unsqueeze(0))
 133             tensor = tensor.unsqueeze(0).unsqueeze(0)
 134             yield tensor.to(
 135                 device=self.device_name,
 136                 dtype=self.compute_dtype if self.device_name.startswith("cuda") else torch.float32,
 137                 non_blocking=True,
 138             )

 140     @torch.inference_mode()
 141     def infer_paths(
 142         self,
 143         paths: list[Path],
 144         *,
 145         output_local_points: bool,
 146         output_world_points: bool,
 147         output_confidence: bool,
 148         dense_output_indices: list[int] | None = None,
 149         image_observer: Callable[[torch.Tensor], None] | None = None,
 150     ) -> dict[str, torch.Tensor]:
 151         self.reset()
 152         self._image_observer = image_observer
 153         output_keys = ["camera_poses"]
 154         if output_local_points or output_world_points:
 155             output_keys.append("local_points")
 156         if output_world_points:
 157             output_keys.append("points")
 158         if output_confidence:
 159             output_keys.append("conf")
 160         autocast_enabled = (
 161             self.device_name.startswith("cuda") and self.compute_dtype != torch.float32
 162         )
 163         device_type = "cuda" if self.device_name.startswith("cuda") else "cpu"
 164         try:
 165             with torch.autocast(
 166                 device_type=device_type,
 167                 dtype=self.compute_dtype,
 168                 enabled=autocast_enabled,
 169             ):
 170                 output = self.network.inference_stream_iter(
 171                     self._frames(paths),
 172                     num_frames=len(paths),
 173                     output_keys=output_keys,
 174                     dense_output_indices=dense_output_indices,
 175                 )
 176         finally:
 177             self._image_observer = None
 178 
 179         def frames_only(value):
 180             if value is None:
 181                 return None
 182             if value.ndim >= 1 and value.shape[0] == 1:
 183                 value = value[0]
 184             return value.detach().float().cpu()
 185 
 186         result = {"camera_poses": frames_only(output["camera_poses"])}
 187         result["attention_backend"] = self.attention_backend
 188         if output_local_points or output_world_points:
 189             result["local_points"] = frames_only(output.get("local_points"))
 190         if output_world_points:
 191             result["world_points"] = frames_only(output.get("points"))
 192         if output_confidence:
 193             logits = frames_only(output.get("conf"))
 194             if logits is None:
 195                 raise RuntimeError("The selected checkpoint has no confidence output")
 196             if logits.ndim == 4 and logits.shape[-1] == 1:
 197                 logits = logits[..., 0]
 198             result["confidence"] = torch.sigmoid(logits)
 199         return result

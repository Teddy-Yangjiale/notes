 213     def _postprocess_stream_carry(
 214         self,
 215         carry: Optional[List[Optional[Any]]],
 216         *,
 217         spatial_hw: Optional[Tuple[int, int]] = None,
 218     ) -> Optional[List[Optional[Any]]]:
 219         """Between stream steps: ``window`` drops whole frames; ``streaming`` packs KV (true VRAM shrink)."""
 220         if carry is None:
 221             return None
 222         mm = str(self.memory_mode).lower()
 223         out: List[Optional[Any]] = []
 224         for kv in carry:
 225             if kv is None:
 226                 out.append(None)
 227                 continue
 228             if isinstance(kv, StreamingKVState):
 229                 if mm != "streaming":
 230                     raise ValueError(
 231                         "StreamingKVState requires memory_mode=streaming"
 232                     )
 233                 # append 已在 stream_chunk_forward 内完成；此处仅 detach。
 234                 out.append(kv.clone_buffers())
 235                 continue
 236             pk, pv = kv[0], kv[1]
 237             pk = pk.contiguous()
 238             pv = pv.contiguous()
 239             T_total = int(pk.shape[2])
 240             if mm == "streaming":
 241                 out.append(
 242                     streaming_carry_from_rect_kv(
 243                         pk,
 244                         pv,
 245                         num_reference_frames=int(self.num_reference_frames),
 246                         local_window_frames=int(self.local_window_frames),
 247                         num_summary_tokens=int(self.streaming_num_summary_tokens),
 248                         max_summary_frames=int(self.paged_max_summary_frames),
 249                     )
 250                 )
 251             elif mm == "window":
 252                 pk2, pv2, _new_t = prune_window_carry_pk_pv(
 253                     pk,
 254                     pv,
 255                     T_total=T_total,
 256                     num_reference_frames=int(self.num_reference_frames),
 257                     local_window_frames=int(self.local_window_frames),
 258                 )
 259                 out.append((pk2, pv2))
 260             else:
 261                 raise ValueError(f"Unknown memory_mode={self.memory_mode!r}")
 262         return out

 273     def _build_pos_grid(
 274         self, B: int, T_total: int, H: int, W: int, device: torch.device
 275     ) -> torch.Tensor:
 276         patch_h, patch_w = H // self.patch_size, W // self.patch_size
 277         pos = self.position_getter(B * T_total, patch_h, patch_w, device)
 278         if self.patch_start_idx > 0:
 279             pos = pos + 1
 280             pos_special = torch.zeros(
 281                 B * T_total, self.patch_start_idx, 2, device=device, dtype=pos.dtype
 282             )
 283             pos = torch.cat([pos_special, pos], dim=1)
 284         hw = pos.shape[1]
 285         return pos.reshape(B, T_total * hw, 2)

 287     def _stream_packed_key_positions(
 288         self,
 289         hs_carry: StreamingKVState,
 290         *,
 291         B: int,
 292         N: int,
 293         H: int,
 294         W: int,
 295         tpf: int,

 314     def _global_attn_stream_packed(
 315         self,
 316         h: torch.Tensor,
 317         pos_q: torch.Tensor,
 318         hs_carry: StreamingKVState,
 319         blk,
 320         attn_mod: StreamingFlashAttention,
 321         B: int,
 322         N: int,
 323         H: int,
 324         W: int,
 325         pos_cache: Dict[str, Any],
 326     ) -> Tuple[torch.Tensor, StreamingKVState]:
 327         """Stream infer: packed streaming carry, no attn bias (see ``infer/stream/attn_sdpa``)."""
 328         tpf = h.shape[1] // N
 329         chunk_k_pos = self._stream_packed_key_positions(
 330             hs_carry, B=B, N=N, H=H, W=W, tpf=tpf, device=h.device, pos_cache=pos_cache
 331         )
 332         xn = blk.norm1(h)
 333         out, new_carry = stream_window_attention(
 334             attn_mod,
 335             xn,
 336             pos_q=pos_q,
 337             hs_carry=hs_carry,
 338             num_new_frames=N,
 339             chunk_key_positions=chunk_k_pos,
 340             image_hw=(H, W),
 341             patch_start_idx=int(self.patch_start_idx),
 342         )
 343         h_out = h + blk.ls1(out)
 344         h_out = h_out + blk.ls2(blk.mlp(blk.norm2(h_out)))

 570                     blk,
 571                     hidden,
 572                     xpos=pos,
 573                     attn_mask=attn_mask,
 574                     past_key_values=None,
 575                     use_cache=False,
 576                     use_reentrant=False,
 577                 )
 578             else:
 579                 hidden = blk(
 580                     hidden,
 581                     xpos=pos,
 582                     attn_mask=attn_mask,
 583                     past_key_values=None,
 584                     use_cache=False,
 585                 )
 586 
 587             if i % 2 == 1:
 588                 global_blk_idx += 1
 589 
 590             if i + 1 in [len(self.decoder) - 1, len(self.decoder)]:
 591                 final_output.append(hidden.reshape(B * N, hw, -1))
 592 
 593         fused = torch.cat([final_output[0], final_output[1]], dim=-1)
 594         pos_out = pos.reshape(B * N, hw, -1)
 595         if long_sequence_parallel and not streaming_inference:
 596             return fused, pos_out, None
 597         return fused, pos_out, past_key_values

 599     def forward(
 600         self,
 601         imgs: torch.Tensor,
 602         causal_global_attn: Optional[bool] = None,
 603         stream_use_cache: bool = False,
 604         past_key_values: Optional[List[Optional[Tuple]]] = None,
 605         ref_hidden: Optional[torch.Tensor] = None,
 606         camera_state: Optional[Dict[str, torch.Tensor]] = None,
 607         long_sequence_parallel: bool = False,
 608         streaming_inference: bool = False,
 609     ) -> Dict[str, Any]:
 610         if causal_global_attn is None:
 611             causal_global_attn = self.causal_global_attn
 612 
 613         imgs = (imgs - self.image_mean) / self.image_std
 614 
 615         B, N, _, H, W = imgs.shape
 616         if long_sequence_parallel:
 617             if stream_use_cache:
 618                 raise ValueError("long_sequence_parallel excludes stream_use_cache")
 619         elif stream_use_cache and N != 1:
 620             raise ValueError("stream_use_cache requires N==1 (single-frame steps with KV cache)")
 621 
 622         patch_h, patch_w = H // 14, W // 14
 623         tks = patch_h * patch_w + self.patch_start_idx
 624 
 625         imgs_bn = imgs.reshape(B * N, _, H, W)
 626         hidden = self.encoder(imgs_bn, is_training=True)
 627 
 628         if isinstance(hidden, dict):
 629             hidden = hidden["x_norm_patchtokens"]
 630 
 631         hidden, pos, pkv = self.decode(
 632             hidden,
 633             N,
 634             H,
 635             W,
 636             causal_global=causal_global_attn,
 637             stream_use_cache=stream_use_cache,
 638             past_key_values=past_key_values,
 639             long_sequence_parallel=long_sequence_parallel,
 640             streaming_inference=streaming_inference,
 641         )
 642 
 643         new_ref_hidden: Optional[torch.Tensor] = None
 644         if self.use_global_points:
 645             if stream_use_cache and N == 1:
 646                 hf = hidden.reshape(B, tks, -1)
 647                 if ref_hidden is None:
 648                     context = hf
 649                     new_ref_hidden = hf.detach()
 650                 else:
 651                     context = ref_hidden
 652                     new_ref_hidden = ref_hidden
 653             else:
 654                 context = (
 655                     hidden.reshape(B, N, tks, -1)[:, 0:1].repeat(1, N, 1, 1).reshape(B * N, tks, -1)
 656                 )
 657             global_point_hidden = self.global_points_decoder(hidden, context, xpos=pos, ypos=pos)
 658 
 659         point_hidden = self.point_decoder(hidden, xpos=pos)
 660         if self.train_conf:
 661             conf_hidden = self.conf_decoder(hidden, xpos=pos)
 662         camera_hidden = self.camera_decoder(hidden, xpos=pos)
 663 
 664         with torch.amp.autocast(device_type="cuda", enabled=False):
 665             point_hidden = point_hidden.float()
 666             ret = self.point_head([point_hidden[:, self.patch_start_idx :]], (H, W)).reshape(
 667                 B, N, H, W, -1
 668             )
 669             xy, z = ret.split([2, 1], dim=-1)
 670             z = depth_from_log_z(z, self.point_z_log_max)
 671             local_points = torch.cat([xy * z, z], dim=-1)
 672 
 673             if self.train_conf:
 674                 conf_hidden = conf_hidden.float()
 675                 conf = self.conf_head([conf_hidden[:, self.patch_start_idx :]], (H, W)).reshape(
 676                     B, N, H, W, -1
 677                 )
 678             else:
 679                 conf = None
 680 
 681             camera_hidden = camera_hidden.float()
 682             camera_tokens_for_head = camera_hidden[:, self.patch_start_idx :]
 683             if (
 684                 self.camera_pose_mode == "relative_adjacent"
 685                 and self.relative_camera_head_type == "token_pair"
 686             ):
 687                 camera_tokens_for_head = camera_hidden
 688             camera_poses, new_camera_state = self._predict_camera_poses(
 689                 camera_tokens_for_head,
 690                 B=B,
 691                 N=N,
 692                 patch_h=patch_h,
 693                 patch_w=patch_w,
 694                 camera_state=camera_state,
 695                 return_state=True,
 696             )
 697 
 698             if self.use_global_points:
 699                 global_point_hidden = global_point_hidden.float()
 700                 global_points = self.global_point_head(
 701                     [global_point_hidden[:, self.patch_start_idx :]], (H, W)
 702                 ).reshape(B, N, H, W, -1)
 703             else:
 704                 global_points = None
 705 
 706             points = torch.einsum(
 707                 "bnij, bnhwj -> bnhwi",
 708                 camera_poses,
 709                 homogenize_points(local_points),
 710             )[..., :3]
 711 
 712         out: Dict[str, Any] = dict(
 713             points=points,
 714             local_points=local_points,
 715             conf=conf,
 716             camera_poses=camera_poses,
 717             global_points=global_points,
 718         )
 719         if stream_use_cache:
 720             out["past_key_values"] = pkv
 721         elif streaming_inference and pkv is not None:
 722             out["past_key_values"] = pkv
 723         if new_ref_hidden is not None:
 724             out["ref_hidden"] = new_ref_hidden
 725         if new_camera_state is not None:
 726             out["camera_state"] = new_camera_state
 727         return out

 729     @staticmethod
 730     def _camera_state_output_slice(
 731         pred: Dict[str, Any], num_frames: int
 732     ) -> Optional[Dict[str, torch.Tensor]]:
 733         state = pred.get("camera_state")
 734         if not isinstance(state, dict):
 735             return None
 736         raw_rel = state.get("raw_adjacent_rel_poses")
 737         source_idx = state.get("source_frame_indices")
 738         if raw_rel is None or source_idx is None:
 739             return None
 740         if raw_rel.shape[1] < num_frames or source_idx.shape[1] < num_frames:
 741             return None
 742         out = {
 743             "raw_adjacent_rel_poses": raw_rel[:, -num_frames:],
 744             "source_frame_indices": source_idx[:, -num_frames:],
 745         }
 746         for bias_key in ("rotation_residual"):
 747             rot_bias = state.get(bias_key)
 748             if rot_bias is not None and rot_bias.numel() > 0:
 749                 if rot_bias.shape[1] == raw_rel.shape[1] - 1:
 750                     pad = rot_bias.new_zeros(rot_bias.shape[0], 1, rot_bias.shape[-1])
 751                     rot_bias = torch.cat([pad, rot_bias], dim=1)
 752                 if rot_bias.shape[1] >= num_frames:
 753                     out[bias_key] = rot_bias[:, -num_frames:]
 754         return out

 756     @staticmethod
 757     def _attach_camera_state_output(
 758         stacked: Dict[str, Any], chunks: List[Optional[Dict[str, torch.Tensor]]]
 759     ) -> Dict[str, Any]:
 760         chunks = [c for c in chunks if c is not None]
 761         if not chunks:
 762             return stacked
 763         metrics = {
 764             "raw_adjacent_rel_poses": torch.cat([c["raw_adjacent_rel_poses"] for c in chunks], dim=1),
 765             "source_frame_indices": torch.cat([c["source_frame_indices"] for c in chunks], dim=1),
 766         }
 767         for bias_key in ("rotation_residual"):
 768             if any(c.get(bias_key) is not None for c in chunks):
 769                 bias_chunks = []
 770                 for c in chunks:
 771                     bias = c.get(bias_key)
 772                     if bias is None:
 773                         raw = c["raw_adjacent_rel_poses"]
 774                         bias = raw.new_zeros(raw.shape[0], raw.shape[1], 3)
 775                     bias_chunks.append(bias)
 776                 metrics[bias_key] = torch.cat(bias_chunks, dim=1)
 777         stacked["camera_state_metrics"] = metrics
 778         return stacked

 809     @torch.inference_mode()
 810     def _inference_stream_sdpa(
 811         self,
 812         imgs: torch.Tensor,
 813         causal_global_attn: Optional[bool] = None,
 814     ) -> Dict[str, Any]:
 815         """Causal streaming with streaming **packed** KV (true VRAM savings) or dense window prune.
 816 
 817         Uses the same causal SDPA mask rules as the paged backend; ``memory_mode=streaming`` summaries mid-memory
 818         frames to the summary register band. ``global_pos_encoding=rope3d`` is supported (gathered
 819         3D RoPE on packed keys).
 820         """
 821         if causal_global_attn is None:
 822             causal_global_attn = self.causal_global_attn
 823         _, N, _, H_img, W_img = imgs.shape
 824         if N < 1:
 825             raise ValueError("_inference_stream_sdpa requires sequence length N >= 1")
 826 
 827         carry: Optional[List[Optional[Tuple]]] = None
 828         ref_h: Optional[torch.Tensor] = None
 829         camera_state: Optional[Dict[str, torch.Tensor]] = None
 830         preds: List[Dict[str, Any]] = []
 831         camera_state_chunks: List[Optional[Dict[str, torch.Tensor]]] = []
 832         _no_pbar = os.environ.get("ABOT_RECON_NO_TQDM", "0") == "1"
 833         for t in tqdm(
 834             range(N),
 835             desc="ABotReconNetwork stream",
 836             unit="frm",
 837             leave=True,
 838             disable=_no_pbar,
 839         ):
 840             frame = imgs[:, t : t + 1]
 841             pred = self.forward(
 842                 frame,
 843                 causal_global_attn=causal_global_attn,
 844                 long_sequence_parallel=True,
 845                 streaming_inference=True,
 846                 past_key_values=carry,
 847                 ref_hidden=ref_h,
 848                 camera_state=camera_state,
 849             )
 850             carry = self._postprocess_stream_carry(
 851                 detach_carry(pred.get("past_key_values")),
 852                 spatial_hw=(int(H_img), int(W_img)),
 853             )
 854             if pred.get("ref_hidden") is not None:
 855                 ref_h = pred["ref_hidden"]
 856             camera_state = pred.get("camera_state", camera_state)
 857             camera_state_chunks.append(self._camera_state_output_slice(pred, frame.shape[1]))
 858             preds.append(
 859                 {
 860                     k: v
 861                     for k, v in pred.items()
 862                     if k not in ("past_key_values", "ref_hidden", "camera_state")
 863                 }
 864             )
 865         stacked: Dict[str, Any] = {}
 866         for k in preds[0].keys():
 867             vals = [p[k] for p in preds]
 868             if vals[0] is None:
 869                 stacked[k] = None
 870             else:
 871                 stacked[k] = torch.cat(vals, dim=1)
 872         return self._attach_camera_state_output(stacked, camera_state_chunks)

 881             return False

 896     def _ensure_paged_manager(
 897         self,
 898         *,
 899         H: int,
 900         W: int,
 901         dtype: torch.dtype,
 902         device: torch.device,
 903     ):
 904         """Create or recreate the per-model paged KV cache manager.
 905 
 906         Rebuilds if the (resolution, dtype, memory_mode, window, reference, summary)
 907         signature changed; otherwise just calls ``reset()``.
 908         """
 909         from abot_recon.modeling.streaming.paged_kv import (
 910             PagedKVCacheManager,
 911         )
 912 
 913         paged_dtype = dtype
 914         if not self.paged_force_fp32 and paged_dtype not in (torch.float16, torch.bfloat16):
 915             # FlashInfer FA2/FA3 kernels do not accept fp32 q/kv dtypes. Validation
 916             # can call stream inference with fp32 images outside autocast, so keep
 917             # the model path usable by storing/running paged attention in bf16.
 918             paged_dtype = torch.bfloat16
 919 
 920         ps = int(self.patch_size)
 921         tpf = (H // ps) * (W // ps) + int(self.patch_start_idx)
 922         # Probe first global attn for head config.
 923         attn0 = self.decoder[1].attn
 924         num_heads = attn0.num_heads
 925         head_dim = attn0.qkv.in_features // num_heads
 926         sig = (
 927             tpf,
 928             num_heads,
 929             head_dim,
 930             int(self.num_reference_frames),
 931             int(self.local_window_frames),
 932             int(self.streaming_num_summary_tokens),
 933             int(self.paged_max_summary_frames),
 934             self.memory_mode,
 935             str(paged_dtype),
 936             str(device),
 937         )
 938         if self._paged_manager is not None and self._paged_manager_sig == sig:
 939             self._paged_manager.reset()
 940             return
 941 
 942         self._paged_manager = PagedKVCacheManager(
 943             num_layers=self.num_global_decoder_blocks,
 944             tpf=tpf,
 945             num_heads=num_heads,
 946             head_dim=head_dim,
 947             num_reference_frames=int(self.num_reference_frames),
 948             local_window_frames=int(self.local_window_frames),
 949             num_summary_tokens=int(self.streaming_num_summary_tokens),
 950             memory_mode=self.memory_mode,
 951             max_total_frames=int(self.paged_max_total_frames),
 952             max_summary_frames=int(self.paged_max_summary_frames),
 953             dtype=paged_dtype,
 954             device=device,
 955             force_fp32=self.paged_force_fp32,
 956         )
 957         self._paged_manager_sig = sig

 959     def _decode_paged(
 960         self,
 961         hidden: torch.Tensor,
 962         *,
 963         H: int,
 964         W: int,
 965         frame_idx: int,
 966     ) -> Tuple[torch.Tensor, torch.Tensor]:
 967         """Single-frame decode (N=1, B=1) using paged KV for global layers.
 968 
 969         Mirrors the ``long_sequence_parallel=True, streaming_inference=True``
 970         slice of ``decode()`` for B=N=1, replacing the global ``stream_chunk_forward``
 971         with ``StreamingFlashAttention.paged_stream_forward``.
 972 
 973         Returns:
 974             (fused_hidden, pos): ``fused`` is the cat of the last two decoder
 975             layers' outputs along the channel dim (matches ``decode()``).
 976         """
 977         manager = self._paged_manager
 978         if manager is None:
 979             raise RuntimeError("paged manager not initialised; call _ensure_paged_manager first")
 980 
 981         B = 1
 982         BN, hw_enc, _ = hidden.shape
 983         if BN != B:
 984             raise ValueError(f"_decode_paged requires B=N=1; got hidden batch={BN}")
 985 
 986         # Prepend register tokens → hidden shape (1, tpf, C)
 987         register_token = self.register_token.repeat(B, 1, 1, 1).reshape(
 988             B, *self.register_token.shape[-2:]
 989         )
 990         hidden = torch.cat([register_token, hidden], dim=1)
 991         hw = hidden.shape[1]
 992 
 993         # Position grid (same layout as decode()): (1, tpf, 2) for pi3_2d
 994         if self.pos_type.startswith("rope"):
 995             pos = self.position_getter(B, H // self.patch_size, W // self.patch_size, hidden.device)
 996         else:
 997             pos = None
 998         if pos is not None and self.patch_start_idx > 0:
 999             pos = pos + 1
1000             pos_special = torch.zeros(
1001                 B, self.patch_start_idx, 2, device=hidden.device, dtype=pos.dtype
1002             )
1003             pos = torch.cat([pos_special, pos], dim=1)
1004         # pos shape: (1, tpf, 2)
1005 
1006         final_output: List[torch.Tensor] = []
1007         global_blk_idx = 0
1008         n_layers = len(self.decoder)
1009 
1010         for i in range(n_layers):
1011             blk = self.decoder[i]
1012             if i % 2 == 0:
1013                 # Local attention: standard block call, hidden in (B*N, hw, C) = (1, tpf, C)
1014                 hidden = blk(
1015                     hidden,
1016                     xpos=pos,
1017                     attn_mask=None,
1018                     past_key_values=None,
1019                     use_cache=False,
1020                 )
1021             else:
1022                 # Global attention via paged KV manager.
1023                 if not isinstance(blk.attn, StreamingFlashAttention):
1024                     raise TypeError(
1025                         f"Expected StreamingFlashAttention at decoder[{i}], "
1026                         f"got {type(blk.attn).__name__}"
1027                     )
1028                 x_norm = blk.norm1(hidden)
1029                 attn_out = blk.attn.paged_stream_forward(
1030                     x_norm,
1031                     pos_q=pos,
1032                     manager=manager,
1033                     layer_idx=global_blk_idx,
1034                     frame_idx=int(frame_idx),
1035                     image_hw=(H, W),
1036                     patch_start_idx=int(self.patch_start_idx),
1037                 )
1038                 hidden = hidden + blk.ls1(attn_out)
1039                 hidden = hidden + blk.ls2(blk.mlp(blk.norm2(hidden)))
1040                 global_blk_idx += 1
1041 
1042             if i + 1 in (n_layers - 1, n_layers):
1043                 final_output.append(hidden.reshape(B, hw, -1))
1044 
1045         fused = torch.cat([final_output[0], final_output[1]], dim=-1)
1046         return fused, pos

1147     def _forward_frame_camera_only(
1148         self,
1149         frame: torch.Tensor,
1150         *,
1151         frame_idx: int,

1154         causal_global_attn: Optional[bool] = None,
1155         past_key_values: Optional[List[Optional[Tuple]]] = None,
1156         use_paged: bool = True,

1159

1161         frame forward, but skips point/conf/global point heads for long pose eval.
1162         """

1164         if causal_global_attn is None:
1165             causal_global_attn = self.causal_global_attn
1166

1168         B, N, _, H, W = imgs.shape
1169         if B != 1 or N != 1:
1170             raise ValueError(f"_forward_frame_camera_only requires B=N=1, got B={B} N={N}")
1171 
1172         patch_h, patch_w = H // 14, W // 14
1173         imgs_bn = imgs.reshape(B * N, _, H, W)

1175         if isinstance(hidden, dict):
1176             hidden = hidden["x_norm_patchtokens"]
1177 
1178         if use_paged:
1179             hidden, pos = self._decode_paged(hidden, H=H, W=W, frame_idx=frame_idx)
1180             pkv = None
1181         else:
1182             hidden, pos, pkv = self.decode(
1183                 hidden,
1184                 N,
1185                 H,
1186                 W,
1187                 causal_global=causal_global_attn,
1188                 stream_use_cache=False,
1189                 past_key_values=past_key_values,
1190                 long_sequence_parallel=True,
1191                 streaming_inference=True,
1192             )
1193 
1194         camera_hidden = self.camera_decoder(hidden, xpos=pos)
1195         with torch.amp.autocast(device_type="cuda", enabled=False):
1196             camera_hidden = camera_hidden.float()
1197             camera_tokens_for_head = camera_hidden[:, self.patch_start_idx :]
1198             if (
1199                 self.camera_pose_mode == "relative_adjacent"
1200                 and self.relative_camera_head_type == "token_pair"
1201             ):
1202                 camera_tokens_for_head = camera_hidden
1203             camera_poses, new_camera_state = self._predict_camera_poses(
1204                 camera_tokens_for_head,
1205                 B=B,
1206                 N=N,
1207                 patch_h=patch_h,
1208                 patch_w=patch_w,
1209                 camera_state=camera_state,

1213                 new_camera_state.pop("_aux_rel_pose_pred", None)
1214                 new_camera_state.pop("_history_rel_pose_pred", None)
1215 
1216         out: Dict[str, Any] = {"camera_poses": camera_poses}
1217         if pkv is not None:
1218             out["past_key_values"] = pkv
1219         if new_camera_state is not None:
1220             out["camera_state"] = new_camera_state
1221         return out

1264             if vals[0] is None:
1265                 stacked[k] = None
1266             else:

1270     @torch.inference_mode()
1271     def inference_stream_iter(
1272         self,
1273         frames: Iterable[torch.Tensor],
1274         *,
1275         num_frames: Optional[int] = None,
1276         causal_global_attn: Optional[bool] = None,
1277         output_keys: Optional[Iterable[str]] = None,
1278         dense_output_indices: Optional[Iterable[int]] = None,
1279     ) -> Dict[str, Any]:
1280         """Streaming inference over already-loaded single-frame tensors.
1281 
1282         Each item must be ``[B,1,3,H,W]``. This avoids staging the full image
1283         sequence on CPU/GPU before inference. Only ``infer_mode=stream`` is
1284         supported because ``full`` mode requires the complete tensor.
1285         """
1286         mode = str(getattr(self, "infer_mode", "full")).lower()
1287         if mode != "stream":
1288             raise ValueError("inference_stream_iter requires infer_mode='stream'")
1289 
1290         iterator = iter(frames)
1291         try:
1292             first = next(iterator)
1293         except StopIteration as exc:
1294             raise ValueError("inference_stream_iter requires at least one frame") from exc
1295 
1296         if first.dim() != 5 or first.shape[1] != 1:
1297             raise ValueError(f"expected frame shape [B,1,3,H,W], got {tuple(first.shape)}")
1298 
1299         allowed = set(output_keys) if output_keys is not None else None
1300         dense_indices = (
1301             None if dense_output_indices is None else {int(index) for index in dense_output_indices}
1302         )
1303         dense_keys = {"points", "local_points", "conf", "global_points"}
1304         collect_camera_state = allowed is None or "camera_state" in allowed
1305         camera_only = allowed is not None and allowed.issubset({"camera_poses", "camera_state"})
1306 
1307         def keep_pred(
1308             pred: Dict[str, Any], *, skip_state: Tuple[str, ...], frame_index: int
1309         ) -> Dict[str, Any]:
1310             return {
1311                 k: v
1312                 for k, v in pred.items()
1313                 if k not in skip_state
1314                 and (allowed is None or k in allowed)
1315                 and (dense_indices is None or k not in dense_keys or frame_index in dense_indices)
1316             }
1317 
1318         def stack_preds(preds: List[Dict[str, Any]]) -> Dict[str, Any]:
1319             if not preds:
1320                 raise ValueError("no predictions were collected; check output_keys")
1321             stacked: Dict[str, Any] = {}
1322             for k in set().union(*(pred.keys() for pred in preds)):
1323                 vals = [pred[k] for pred in preds if k in pred]
1324                 if vals[0] is None:
1325                     stacked[k] = None
1326                 else:
1327                     stacked[k] = torch.cat(vals, dim=1)
1328             return stacked
1329 
1330         frames_iter = chain((first,), iterator)
1331         total = num_frames if num_frames is not None else None
1332         _no_pbar = os.environ.get("ABOT_RECON_NO_TQDM", "0") == "1"
1333 
1334         if self._can_use_paged_kv(first):
1335             B, _, _, H_img, W_img = first.shape
1336             if B != 1:
1337                 raise ValueError(f"paged inference_stream_iter requires B=1, got {B}")
1338             self._ensure_paged_manager(H=H_img, W=W_img, dtype=first.dtype, device=first.device)
1339             ref_h: Optional[torch.Tensor] = None
1340             camera_state: Optional[Dict[str, torch.Tensor]] = None
1341             preds: List[Dict[str, Any]] = []
1342             camera_state_chunks: List[Optional[Dict[str, torch.Tensor]]] = []
1343             for t, frame in enumerate(
1344                 tqdm(
1345                     frames_iter,
1346                     total=total,
1347                     desc="ABotReconNetwork paged stream",
1348                     unit="frm",
1349                     leave=True,
1350                     disable=_no_pbar,
1351                 )
1352             ):
1353                 frame_camera_only = camera_only or (
1354                     dense_indices is not None and t not in dense_indices
1355                 )
1356                 if frame_camera_only:
1357                     pred = self._forward_frame_camera_only(
1358                         frame,
1359                         frame_idx=t,
1360                         ref_hidden=ref_h,
1361                         camera_state=camera_state,
1362                         use_paged=True,
1363                     )
1364                 else:
1365                     pred = self._forward_frame_paged(
1366                         frame, frame_idx=t, ref_hidden=ref_h, camera_state=camera_state
1367                     )
1368                 if pred.get("ref_hidden") is not None:
1369                     ref_h = pred["ref_hidden"]
1370                 camera_state = pred.get("camera_state", camera_state)
1371                 if collect_camera_state:
1372                     camera_state_chunks.append(
1373                         self._camera_state_output_slice(pred, frame.shape[1])
1374                     )
1375                 preds.append(
1376                     keep_pred(
1377                         pred,
1378                         skip_state=("ref_hidden", "camera_state"),
1379                         frame_index=t,
1380                     )
1381                 )
1382             out = stack_preds(preds)
1383             return (
1384                 self._attach_camera_state_output(out, camera_state_chunks)
1385                 if collect_camera_state
1386                 else out
1387             )
1388 
1389         if causal_global_attn is None:
1390             causal_global_attn = self.causal_global_attn
1391         _, _, _, H_img, W_img = first.shape
1392         carry: Optional[List[Optional[Tuple]]] = None
1393         ref_h: Optional[torch.Tensor] = None
1394         camera_state: Optional[Dict[str, torch.Tensor]] = None
1395         preds: List[Dict[str, Any]] = []
1396         camera_state_chunks: List[Optional[Dict[str, torch.Tensor]]] = []
1397         for frame_index, frame in enumerate(
1398             tqdm(
1399                 frames_iter,
1400                 total=total,
1401                 desc="ABotReconNetwork stream",
1402                 unit="frm",
1403                 leave=True,
1404                 disable=_no_pbar,
1405             )
1406         ):
1407             frame_camera_only = camera_only or (
1408                 dense_indices is not None and frame_index not in dense_indices
1409             )
1410             if frame_camera_only:
1411                 pred = self._forward_frame_camera_only(
1412                     frame,
1413                     frame_idx=frame_index,
1414                     ref_hidden=ref_h,
1415                     camera_state=camera_state,
1416                     causal_global_attn=causal_global_attn,
1417                     past_key_values=carry,
1418                     use_paged=False,
1419                 )
1420             else:
1421                 pred = self.forward(
1422                     frame,
1423                     causal_global_attn=causal_global_attn,
1424                     long_sequence_parallel=True,
1425                     streaming_inference=True,
1426                     past_key_values=carry,
1427                     ref_hidden=ref_h,
1428                     camera_state=camera_state,
1429                 )
1430             carry = self._postprocess_stream_carry(
1431                 detach_carry(pred.get("past_key_values")),
1432                 spatial_hw=(int(H_img), int(W_img)),
1433             )
1434             if pred.get("ref_hidden") is not None:
1435                 ref_h = pred["ref_hidden"]
1436             camera_state = pred.get("camera_state", camera_state)
1437             if collect_camera_state:
1438                 camera_state_chunks.append(self._camera_state_output_slice(pred, frame.shape[1]))
1439             preds.append(
1440                 keep_pred(
1441                     pred,
1442                     skip_state=("past_key_values", "ref_hidden", "camera_state"),
1443                     frame_index=frame_index,
1444                 )
1445             )
1446         out = stack_preds(preds)
1447         return (
1448             self._attach_camera_state_output(out, camera_state_chunks)
1449             if collect_camera_state
1450             else out
1451         )

1453     def _load_deferred_streaming_checkpoint(self, ckpt: str) -> None:
1454         """Load pretrained weights after all ABot-Recon-specific modules exist.
1455 
1456         The base Pi3 constructor cannot load ABot-Recon checkpoints safely because
1457         gate modules are registered in this subclass after ``super().__init__``.
1458         Keeping this loader here makes ``model.ckpt`` safe for training and eval.
1459         """
1460         ckpt_path = str(ckpt)
1461         src = (
1462             "OSS (buffered in RAM via ossutil.open)"
1463             if ckpt_path.startswith("oss://")
1464             else "local/NAS path"
1465         )
1466         print(
1467             f"[ABotReconNetwork] Reading deferred checkpoint ({src}; can be slow)…\\n"
1468             f"      path={ckpt_path}",
1469             flush=True,
1470         )
1471         checkpoint = _load_raw_pi3_checkpoint(ckpt_path)
1472         if isinstance(checkpoint, dict):
1473             if "state_dict" in checkpoint:
1474                 checkpoint = checkpoint["state_dict"]
1475             elif "model" in checkpoint and isinstance(checkpoint["model"], dict):
1476                 checkpoint = checkpoint["model"]
1477 
1478         checkpoint_has_conf_decoder = any(
1479             key.startswith("conf_decoder.") or ".conf_decoder." in key for key in checkpoint.keys()
1480         )
1481         print("[ABotReconNetwork] Applying deferred state_dict to complete streaming model…", flush=True)
1482         res = self.load_state_dict(checkpoint, strict=False)
1483         if (
1484             self.enable_confidence
1485             and self.init_conf_decoder_from_point
1486             and not checkpoint_has_conf_decoder
1487         ):
1488             self._initialize_conf_decoder_from_point()
1489         missing = list(getattr(res, "missing_keys", []))
1490         unexpected = list(getattr(res, "unexpected_keys", []))
1491         gate_bad = [k for k in missing + unexpected if ".gate_proj." in k]
1492         if gate_bad:
1493             preview = ", ".join(gate_bad[:16])
1494             raise RuntimeError(
1495                 "Streaming checkpoint gate weights were not loaded cleanly after deferred load: "
1496                 f"{preview}. Check model.gate_layers and checkpoint compatibility."
1497             )
1498         print(f"[ABotReconNetwork] Load checkpoints from {ckpt}: {res}", flush=True)
1499         del checkpoint
1500         if torch.cuda.is_available():
1501             torch.cuda.empty_cache()

 211 def cam_pose(n, delta=None):
 212     M = C2Wt[n]
 213     d = delta if delta is not None else (dpose[n] if dpose is not None else None)
 214     if d is None: return M
 215     w = d[:3] * POSE_LR[:3]; t = d[3:] * POSE_LR[3:]; z = torch.zeros((), device=dev)
 216     Kx = torch.stack([torch.stack([z, -w[2], w[1]]), torch.stack([w[2], z, -w[0]]), torch.stack([-w[1], w[0], z])])
 217     Dm = torch.eye(4, device=dev); Dm = Dm.clone(); Dm[:3, :3] = torch.linalg.matrix_exp(Kx); Dm[:3, 3] = t
 218     return M @ Dm
 219 def render(c2w, mode="RGB+ED"):
 220     r, a, info = rasterization(P["means"], P["quats"], torch.exp(P["scales"]), torch.sigmoid(P["opacities"]), P["sh0"], torch.linalg.inv(c2w)[None], Kt[None],
 221                                W_, H_, near_plane=0.3, far_plane=600.0, sh_degree=0, packed=False, render_mode=mode)
 222     return r[0], a[0, ..., 0], info
 223 _win = None
 224 def ssim(x, y):
 225     global _win
 226     if _win is None:
 227         g = torch.tensor([math.exp(-(i - 5) ** 2 / (2 * 1.5 ** 2)) for i in range(11)], device=x.device); g = g / g.sum()
 228         _win = (g[:, None] * g[None, :])[None, None].repeat(3, 1, 1, 1)
 229     mx = F.conv2d(x, _win, padding=5, groups=3); my = F.conv2d(y, _win, padding=5, groups=3)
 230     sxx = F.conv2d(x * x, _win, padding=5, groups=3) - mx * mx; syy = F.conv2d(y * y, _win, padding=5, groups=3) - my * my; sxy = F.conv2d(x * y, _win, padding=5, groups=3) - mx * my
 231     return (((2 * mx * my + 1e-4) * (2 * sxy + 9e-4)) / ((mx * mx + my * my + 1e-4) * (sxx + syy + 9e-4))).mean()

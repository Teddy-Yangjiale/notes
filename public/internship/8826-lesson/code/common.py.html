 115 def road_frames(poses, smooth_xy=4.0, smooth_h=40.0, max_grade=0.08):
 116     """由融合位姿的**位置**重建逐帧"贴路"坐标架 (ENU 标注, 数组列顺序 E,U,N)。
 117 
 118     只信位置、不信朝向 —— 2026-10-01 核实: 融合位姿的朝向在约三成行驶帧上与轨迹切向差 10 度以上
 119     (第 3100 帧附近: 位姿朝向 226 度, 轨迹切向 93 度, 车速 56 km/h), 与加速度计倾角无关,
 120     是位姿图里朝向只受弱约束 (相对旋转 sigma 15 度) 落进了错误解。位置则有 GNSS 直接约束 (残差中位 1.45 m)。
 121 
 122     做法:
 123       位置   按里程重采样后做高斯平滑 (sigma=4 m) —— 车辆轨迹本身没有高频抖动, 抖动来自定位噪声
 124              在相邻节点间的线性摊派; 平滑后相邻帧的地面条带才接得上;
 125       朝向   取平滑轨迹的切向 (即行进方向 = 直行时的车道方向);
 126       高程   按里程 1 m 分箱取中位 (消掉停车漂移) 再平滑 sigma=40 m; 相机高与纵坡取自同一条剖面。
 127     返回:
 128        p_orig 融合位姿原始位置 (反算相机系坐标时用)
 129        p      平滑后的相机位置;  fwd_h/right  水平前进方向/地理右侧;  fwd/down  沿坡前向/垂直路面向下
 130        gamma  纵坡 (rad);  s  累计里程;  moving  该帧是否在动;  health  0~1, 该帧几何可信度
 131     """
 132     P = np.asarray(poses, np.float64)
 133     p_fused = P[:, :3, 3].copy()
 134     n = len(p_fused)
 135     # 录像帧的真实时刻 != 帧号/5.827: 见 frame_lag()。融合位姿是按名义时刻取的定位,
 136     # 所以第 i 帧真正所在的位置 = 融合轨迹上"名义时刻 + 时延"处的位置, 即把轨迹按帧号平移 lag*FPS。
 137     lag = frame_lag(n)
 138     src = np.clip(np.arange(n) + lag * FPS, 0, n - 1)
 139     p0 = np.stack([np.interp(src, np.arange(n), p_fused[:, c]) for c in range(3)], 1)
 140     step0 = np.linalg.norm(np.diff(p0[:, [0, 2]], axis=0), axis=1)
 141     s0 = np.concatenate([[0], np.cumsum(step0)])
 142     grid = np.arange(0, s0[-1] + 0.5, 0.5)
 143     # 里程是单调不减的; 停车帧共享同一里程, 重采样时自然合并
 144     eu = np.interp(grid, s0, p0[:, 0]); nu = np.interp(grid, s0, p0[:, 2])
 145     # 平滑长度: 直路上取 15 m —— 实测 (build/probe_saw.py) 同一条车道线在相邻 IPM 条带间的横向错位
 146     # 从 sigma=4 m 时的 7.2 cm 降到 4.0 cm; 弯道处退回 4 m, 否则会把弯切掉 (切弯量约 sigma^2 / 2R)。
 147     e4, n4 = _gauss(eu, smooth_xy / 0.5), _gauss(nu, smooth_xy / 0.5)
 148     e15, n15 = _gauss(eu, 15.0 / 0.5), _gauss(nu, 15.0 / 0.5)
 149     hd4 = np.unwrap(np.arctan2(np.gradient(e4), np.gradient(n4)))
 150     curv = np.abs(np.degrees(np.gradient(_gauss(hd4, 8.0 / 0.5), 0.5)))        # deg / m
 151     wst = np.clip(1.0 - (_gauss(curv, 12.0 / 0.5) - 0.25) / 0.5, 0.0, 1.0)     # 1 = 直路, 0 = 弯道
 152     wst = _gauss(wst, 8.0 / 0.5)
 153     es, ns = wst * e15 + (1 - wst) * e4, wst * n15 + (1 - wst) * n4
 154     b = np.clip((s0 / 0.5).astype(int), 0, len(grid) - 1)
 155     hu = np.full(len(grid), np.nan)
 156     order = np.argsort(b, kind="stable"); cut = np.flatnonzero(np.diff(b[order])) + 1
 157     for seg in np.split(order, cut):
 158         hu[b[seg[0]]] = np.median(p0[seg, 1])
 159     ok = np.isfinite(hu); hu = np.interp(grid, grid[ok], hu[ok])
 160     hs = _gauss(hu, smooth_h / 0.5)
 161     grad = np.clip(np.gradient(hs, 0.5), -max_grade, max_grade)
 162     te, tn = np.gradient(es, 0.5), np.gradient(ns, 0.5)
 163     tl = np.hypot(te, tn); tl[tl < 1e-6] = 1.0
 164     te, tn = te / tl, tn / tl
 165     p = np.stack([np.interp(s0, grid, es), np.interp(s0, grid, hs), np.interp(s0, grid, ns)], 1)
 166     fh = np.stack([np.interp(s0, grid, te), np.zeros(n), np.interp(s0, grid, tn)], 1)
 167     fh /= np.linalg.norm(fh, axis=1, keepdims=True)
 168     right = np.stack([fh[:, 2], np.zeros(n), -fh[:, 0]], 1)          # (E,N) 顺时针转 90 度
 169     gamma = np.arctan(np.interp(s0, grid, grad))
 170     up = np.array([0.0, 1.0, 0.0])
 171     cg, sg = np.cos(gamma)[:, None], np.sin(gamma)[:, None]
 172     fwd = fh * cg + up * sg
 173     down = -up * cg + fh * sg
 174     moving = np.r_[step0, step0[-1]] > 0.15
 175     # 可信度: 平滑轨迹的曲率 (急弯处切向不稳) + 平滑前后位置差 (定位跳变) + 原位姿朝向与切向的夹角
 176     f0 = P[:, :3, 2][:, [0, 2]]; f0 /= np.linalg.norm(f0, axis=1, keepdims=True)
 177     cosd = np.clip((f0 * fh[:, [0, 2]]).sum(1), -1, 1)
 178     dpos = np.maximum(np.hypot(p[:, 0] - p0[:, 0], p[:, 2] - p0[:, 2]) - 0.6, 0.0)   # 平滑本身带来的 0.6 m 以内不算毛病
 179     turn = np.abs(np.degrees(np.gradient(np.unwrap(np.arctan2(te, tn)), 0.5)))       # deg / m
 180     turn_f = np.interp(s0, grid, _gauss(turn, 6.0))
 181     health = np.clip(1.0 - dpos / 2.0, 0, 1) * np.clip(1.0 - turn_f / 2.5, 0, 1)
 182     return dict(p_orig=p_fused, p_lagfix=p0, lag=lag, p=p, fwd_h=fh, right=right, fwd=fwd, down=down, gamma=gamma, s=s0,
 183                 moving=moving, health=health, pose_heading_dev=np.degrees(np.arccos(cosd)), smooth_shift=dpos)

 186 def cam_to_vehicle():
 187     """相机系 (x 右, y 下, z 前) -> 车体/路面系 (X 右, Y 前, Z 上) 的旋转, 与 bev_core.BEVRenderer._rot 同一约定。
 188     横滚取 0: 稠密帧路面平面拟合给出相机相对路面的横滚中位 0.06 度 (标定标称 1.80 度没有被数据支持)。"""
 189     r, p, y = 0.0, math.radians(CAM_PITCH_UP), math.radians(CAM_YAW)
 190     Rz = np.array([[np.cos(y), -np.sin(y), 0], [np.sin(y), np.cos(y), 0], [0, 0, 1]])
 191     Rx = np.array([[1, 0, 0], [0, np.cos(p), -np.sin(p)], [0, np.sin(p), np.cos(p)]])
 192     Ry = np.array([[np.cos(r), 0, np.sin(r)], [0, 1, 0], [-np.sin(r), 0, np.cos(r)]])
 193     base = np.array([[1, 0, 0], [0, 0, 1], [0, -1, 0]], float)
 194     return Rz @ Rx @ Ry @ base

 197 def reproject(xyz, frame, poses, rf, chunk=4_000_000):
 198     """把 ~/gd 现成点云里的点改投到地理真实的位置。xyz: (N,3) 列序 E,U,N (原样), frame: 来源帧。
 199     先用原位姿把点**精确**还原到相机系 (这一步与原位姿好坏无关, 只是求逆), 再用贴路坐标架重新放置。"""
 200     P = np.asarray(poses, np.float64)
 201     Rp = P[:, :3, :3]
 202     A = cam_to_vehicle()
 203     out = np.empty_like(xyz, dtype=np.float32)
 204     for a in range(0, len(xyz), chunk):
 205         b = min(len(xyz), a + chunk)
 206         F = frame[a:b]
 207         rel = xyz[a:b].astype(np.float64) - rf["p_orig"][F]
 208         pc = np.einsum("nji,nj->ni", Rp[F], rel) * MODEL_SHRINK      # R'^T (X - p), 再绕相机中心收缩
 209         pv = pc @ A.T                                                # (X 右, Y 前, Z 上)
 210         out[a:b] = (rf["p"][F] + rf["right"][F] * pv[:, [0]] + rf["fwd"][F] * pv[:, [1]]
 211                     - rf["down"][F] * pv[:, [2]]).astype(np.float32)
 212     return out

 215 class RoadField:
 216     """路面高程场: 任意 (E,N) -> 路面高 U。取最近的行驶帧, 相机高减去 IPM_HEIGHT, 再按纵坡外推。横坡不计。"""
 217 
 218     def __init__(self, frames):
 219         from scipy.spatial import cKDTree
 220         mv = np.flatnonzero(frames["moving"])
 221         self.idx = mv
 222         self.p = frames["p"][mv]; self.fh = frames["fwd_h"][mv]; self.g = np.tan(frames["gamma"][mv])
 223         self.tree = cKDTree(self.p[:, [0, 2]])
 224 
 225     def __call__(self, e, n, return_index=False):
 226         q = np.stack([np.asarray(e, float).ravel(), np.asarray(n, float).ravel()], 1)
 227         d, k = self.tree.query(q)
 228         lon = (q[:, 0] - self.p[k, 0]) * self.fh[k, 0] + (q[:, 1] - self.p[k, 2]) * self.fh[k, 2]
 229         u = self.p[k, 1] - IPM_HEIGHT + self.g[k] * lon
 230         if return_index:
 231             return u.reshape(np.shape(e)), self.idx[k].reshape(np.shape(e)), d.reshape(np.shape(e))
 232         return u.reshape(np.shape(e))

 251         x, y, bw, bh, area = st[i]
 252         if area > max_hole or x == 0 or y == 0 or x + bw >= w or y + bh >= h: continue
 253         m = cc[y:y + bh, x:x + bw] == i
 254         ring = cv2.dilate(m.astype(np.uint8), np.ones((3, 3), np.uint8), borderType=cv2.BORDER_CONSTANT)
 255         y0, x0 = max(0, y - 1), max(0, x - 1)
 256         box = lab[y0:y + bh + 1, x0:x + bw + 1]
 257         mm = cv2.dilate((cc[y0:y + bh + 1, x0:x + bw + 1] == i).astype(np.uint8), np.ones((3, 3), np.uint8)) > 0
 258         nb = box[mm & (cc[y0:y + bh + 1, x0:x + bw + 1] != i)]
 259         if len(nb) and (nb == 1).mean() > 0.9: out[y:y + bh, x:x + bw][m] = 1
 260     return out

 263 def load_label(seg_dir, stem, mark_dir=None):
 264     """读一帧的 SAM 3 标签 (0 其他 / 1 路面 / 2 车辆), 并把"标线"掩膜并进路面。
 265 
 266     SAM 3 的 "road" 会把近处的大块白色标线 (箭头、近处的虚线, 后摄俯视角度下几乎所有白线) 当成"不是路面"抠出去。
 267     标线是路面的一部分, 也是配准的主要依据, 所以另跑一遍 "road marking" (t00_seg_mark.py), 挨着路面的标线连通块整块算作路面。
 268     找不到文件返回 None。"""
 269     import cv2
 270     lab = cv2.imread("%s/%s.png" % (seg_dir, stem), 0)
 271     if lab is None: return None
 272     if mark_dir:
 273         mark = cv2.imread("%s/%s.png" % (mark_dir, stem), 0)
 274         if mark is not None and mark.shape == lab.shape:
 275             cand = ((mark > 0) & (lab == 0)).astype(np.uint8)
 276             if cand.any():
 277                 near_road = cv2.dilate((lab == 1).astype(np.uint8), np.ones((9, 9), np.uint8)) > 0
 278                 ncc, cc = cv2.connectedComponents(cand, connectivity=8)
 279                 touch = np.unique(cc[near_road & (cand > 0)])
 280                 lab = lab.copy(); lab[np.isin(cc, touch[touch > 0])] = 1
 281     return clean_label(lab, max_hole=400)

 284 def vehicle_mask(lab, grow=5, down=14):
 285     """车辆/行人掩膜 (标签 2) 外扩: 四周 grow 像素, 向下再多 down 像素。
 286     向下多扩是为了盖住车底的阴影 —— 阴影在分割里算"路面", 但它跟着车走, 贴进正射图就是沿车道的一串暗斑。
 287     lab 是 960x540 (或同比例) 的标签图; 返回 bool。"""
 288     import cv2
 289     v = (lab == 2).astype(np.uint8)
 290     if not v.any(): return v.astype(bool)
 291     out = cv2.dilate(v, np.ones((grow, grow), np.uint8))
 292     if down > 0:
 293         k = np.ones((down + 1, 5), np.uint8)
 294         out |= cv2.dilate(v, k, anchor=(2, down))          # 锚点放在核的最下一行: 结果向下延伸 down 个像素
 295     return out > 0

  34 def _to_chw01(image: Image.Image | np.ndarray | torch.Tensor) -> torch.Tensor:
  35     if isinstance(image, Image.Image):
  36         return tvf.to_tensor(image.convert("RGB"))
  37     if isinstance(image, np.ndarray):
  38         tensor = torch.from_numpy(np.asarray(image).copy())
  39         if tensor.ndim != 3 or tensor.shape[-1] != 3:
  40             raise ValueError(f"Expected HWC RGB array, got {tuple(tensor.shape)}")
  41         tensor = tensor.permute(2, 0, 1)
  42     elif torch.is_tensor(image):
  43         tensor = image.detach().cpu()
  44         if tensor.ndim != 3:
  45             raise ValueError(f"Expected three-dimensional image, got {tuple(tensor.shape)}")
  46         if tensor.shape[-1] == 3:
  47             tensor = tensor.permute(2, 0, 1)
  48         elif tensor.shape[0] != 3:
  49             raise ValueError(f"Expected CHW or HWC RGB tensor, got {tuple(tensor.shape)}")
  50     else:
  51         raise TypeError(f"Unsupported image type: {type(image).__name__}")
  52     tensor = tensor.float()
  53     if tensor.numel() and float(tensor.max()) > 1.5:
  54         tensor = tensor / 255.0
  55     return tensor.clamp_(0.0, 1.0).contiguous()

  58 def preprocess_image(
  59     image: Image.Image | np.ndarray | torch.Tensor,
  60     *,
  61     height: int = 280,
  62     width: int = 504,
  63     pad_rgb: Sequence[float] = DEFAULT_PAD_RGB,
  64 ) -> tuple[torch.Tensor, FovTransform]:
  65     """Width-lock resize followed by center crop or ImageNet-mean vertical pad."""
  66     tensor = _to_chw01(image)
  67     _, source_height, source_width = tensor.shape
  68     resized_height = max(1, round(source_height * width / max(source_width, 1)))
  69     tensor = tvf.resize(
  70         tensor,
  71         [resized_height, width],
  72         interpolation=InterpolationMode.BICUBIC,
  73         antialias=True,
  74     )
  75     crop_top = crop_bottom = pad_top = pad_bottom = 0
  76     if resized_height > height:
  77         crop_top = round((resized_height - height) * 0.5)
  78         crop_bottom = resized_height - height - crop_top
  79         tensor = tvf.crop(tensor, crop_top, 0, height, width)
  80     elif resized_height < height:
  81         pad_top = (height - resized_height) // 2
  82         pad_bottom = height - resized_height - pad_top
  83         color = torch.as_tensor(pad_rgb, dtype=tensor.dtype).view(3, 1, 1)
  84         canvas = color.expand(3, height, width).clone()
  85         canvas[:, pad_top : pad_top + resized_height] = tensor
  86         tensor = canvas
  87     transform = FovTransform(
  88         source_height=source_height,
  89         source_width=source_width,
  90         resized_height=resized_height,
  91         target_height=height,
  92         target_width=width,
  93         crop_top=crop_top,
  94         crop_bottom=crop_bottom,
  95         pad_top=pad_top,
  96         pad_bottom=pad_bottom,
  97     )
  98     # Preserve the tiny bicubic boundary overshoots used by the published
  99     # inference pipeline; clamping here changes long-horizon pose composition.
 100     return tensor, transform

 103 def iter_preprocessed(
 104     paths: Iterable[str | Path], *, height: int = 280, width: int = 504
 105 ):
 106     reference: FovTransform | None = None
 107     for index, path in enumerate(paths):
 108         with Image.open(path) as image:
 109             tensor, transform = preprocess_image(image, height=height, width=width)
 110         if reference is None:
 111             reference = transform
 112         elif transform != reference:
 113             raise ValueError(
 114                 f"Inconsistent frame geometry at index {index}: {transform} != {reference}"
 115             )
 116         yield tensor, transform

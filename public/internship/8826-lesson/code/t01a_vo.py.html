  41 def rigid_fit(PB, PA, w=None):
  42     """(加权) 最小二乘刚体: PA ~ R PB + t"""
  43     if w is None: w = np.ones(len(PA))
  44     w = w / w.sum()
  45     cb, ca = (PB * w[:, None]).sum(0), (PA * w[:, None]).sum(0)
  46     H = ((PB - cb) * w[:, None]).T @ (PA - ca)
  47     th = np.arctan2(H[0, 1] - H[1, 0], H[0, 0] + H[1, 1])
  48     c, s = np.cos(th), np.sin(th); R = np.array([[c, -s], [s, c]])
  49     return th, ca - R @ cb

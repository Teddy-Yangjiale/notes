  14 def _multinomial_sample(weights: Tensor, n: int, replacement: bool = True) -> Tensor:
  15     """Sample from a distribution using torch.multinomial or numpy.random.choice.
  16 
  17     This function adaptively chooses between `torch.multinomial` and `numpy.random.choice`
  18     based on the number of elements in `weights`. If the number of elements exceeds
  19     the torch.multinomial limit (2^24), it falls back to using `numpy.random.choice`.
  20 
  21     Args:
  22         weights (Tensor): A 1D tensor of weights for each element.
  23         n (int): The number of samples to draw.
  24         replacement (bool): Whether to sample with replacement. Default is True.
  25 
  26     Returns:
  27         Tensor: A 1D tensor of sampled indices.
  28     """
  29     num_elements = weights.size(0)
  30 
  31     if num_elements <= 2**24:
  32         # Use torch.multinomial for elements within the limit
  33         return torch.multinomial(weights, n, replacement=replacement)
  34     else:
  35         # Fallback to numpy.random.choice for larger element spaces
  36         weights = weights / weights.sum()
  37         weights_np = weights.detach().cpu().numpy()
  38         sampled_idxs_np = np.random.choice(
  39             num_elements, size=n, p=weights_np, replace=replacement
  40         )
  41         sampled_idxs = torch.from_numpy(sampled_idxs_np)
  42 
  43         # Return the sampled indices on the original device
  44         return sampled_idxs.to(weights.device)

  47 @torch.no_grad()
  48 def _update_param_with_optimizer(
  49     param_fn: Callable[[str, Tensor], Tensor],
  50     optimizer_fn: Callable[[str, Tensor], Tensor],
  51     params: Union[Dict[str, torch.nn.Parameter], torch.nn.ParameterDict],
  52     optimizers: Dict[str, torch.optim.Optimizer],
  53     names: Union[List[str], None] = None,
  54 ):
  55     """Update the parameters and the state in the optimizers with defined functions.
  56 
  57     Args:
  58         param_fn: A function that takes the name of the parameter and the parameter itself,
  59             and returns the new parameter.
  60         optimizer_fn: A function that takes the key of the optimizer state and the state value,
  61             and returns the new state value.
  62         params: A dictionary of parameters.
  63         optimizers: A dictionary of optimizers, each corresponding to a parameter.
  64         names: A list of key names to update. If None, update all. Default: None.
  65     """
  66     if names is None:
  67         # If names is not provided, update all parameters
  68         names = list(params.keys())
  69 
  70     for name in names:
  71         param = params[name]
  72         new_param = param_fn(name, param)
  73         params[name] = new_param
  74         if name not in optimizers:
  75             assert not param.requires_grad, (
  76                 f"Optimizer for {name} is not found, but the parameter is trainable."
  77                 f"Got requires_grad={param.requires_grad}"
  78             )
  79             continue
  80         optimizer = optimizers[name]
  81         for i in range(len(optimizer.param_groups)):
  82             param_state = optimizer.state[param]
  83             del optimizer.state[param]
  84             for key in param_state.keys():
  85                 if key != "step":
  86                     v = param_state[key]
  87                     param_state[key] = optimizer_fn(key, v)
  88             optimizer.param_groups[i]["params"] = [new_param]
  89             optimizer.state[new_param] = param_state

 100 
 101     Args:
 102         params: A dictionary of parameters.
 103         optimizers: A dictionary of optimizers, each corresponding to a parameter.
 104         mask: A boolean mask to duplicate the Gaussians.
 105     """
 106     device = mask.device
 107     sel = torch.where(mask)[0]
 108

 110         return torch.nn.Parameter(torch.cat([p, p[sel]]), requires_grad=p.requires_grad)
 111

 113         return torch.cat([v, torch.zeros((len(sel), *v.shape[1:]), device=device)])

 116     _update_param_with_optimizer(param_fn, optimizer_fn, params, optimizers)
 117     # update the extra running state
 118     for k, v in state.items():

 123 @torch.no_grad()
 124 def split(
 125     params: Union[Dict[str, torch.nn.Parameter], torch.nn.ParameterDict],
 126     optimizers: Dict[str, torch.optim.Optimizer],
 127     state: Dict[str, Tensor],
 128     mask: Tensor,
 129     revised_opacity: bool = False,
 130 ):
 131     """Inplace split the Gaussian with the given mask.
 132 
 133     Args:
 134         params: A dictionary of parameters.
 135         optimizers: A dictionary of optimizers, each corresponding to a parameter.
 136         mask: A boolean mask to split the Gaussians.
 137         revised_opacity: Whether to use revised opacity formulation
 138           from arXiv:2404.06109. Default: False.
 139     """
 140     device = mask.device
 141     sel = torch.where(mask)[0]
 142     rest = torch.where(~mask)[0]
 143 
 144     scales = torch.exp(params["scales"][sel])
 145     quats = F.normalize(params["quats"][sel], dim=-1)
 146     rotmats = normalized_quat_to_rotmat(quats)  # [N, 3, 3]
 147     samples = torch.einsum(
 148         "nij,nj,bnj->bni",
 149         rotmats,
 150         scales,
 151         torch.randn(2, len(scales), 3, device=device),
 152     )  # [2, N, 3]
 153 
 154     def param_fn(name: str, p: Tensor) -> Tensor:
 155         repeats = [2] + [1] * (p.dim() - 1)
 156         if name == "means":
 157             p_split = (p[sel] + samples).reshape(-1, 3)  # [2N, 3]
 158         elif name == "scales":
 159             p_split = torch.log(scales / 1.6).repeat(2, 1)  # [2N, 3]
 160         elif name == "opacities" and revised_opacity:
 161             new_opacities = 1.0 - torch.sqrt(1.0 - torch.sigmoid(p[sel]))
 162             p_split = torch.logit(new_opacities).repeat(repeats)  # [2N]
 163         else:
 164             p_split = p[sel].repeat(repeats)
 165         p_new = torch.cat([p[rest], p_split])
 166         p_new = torch.nn.Parameter(p_new, requires_grad=p.requires_grad)
 167         return p_new
 168 
 169     def optimizer_fn(key: str, v: Tensor) -> Tensor:
 170         v_split = torch.zeros((2 * len(sel), *v.shape[1:]), device=device)
 171         return torch.cat([v[rest], v_split])
 172 
 173     # update the parameters and the state in the optimizers
 174     _update_param_with_optimizer(param_fn, optimizer_fn, params, optimizers)
 175     # update the extra running state
 176     for k, v in state.items():
 177         if isinstance(v, torch.Tensor):
 178             repeats = [2] + [1] * (v.dim() - 1)
 179             v_new = v[sel].repeat(repeats)
 180             state[k] = torch.cat((v[rest], v_new))

 244 @torch.no_grad()
 245 def relocate(
 246     params: Union[Dict[str, torch.nn.Parameter], torch.nn.ParameterDict],
 247     optimizers: Dict[str, torch.optim.Optimizer],
 248     state: Dict[str, Tensor],
 249     mask: Tensor,
 250     binoms: Tensor,
 251     min_opacity: float = 0.005,
 252 ):
 253     """Inplace relocate some dead Gaussians to the lives ones.
 254 
 255     Args:
 256         params: A dictionary of parameters.
 257         optimizers: A dictionary of optimizers, each corresponding to a parameter.
 258         mask: A boolean mask to indicates which Gaussians are dead.
 259     """
 260     # support "opacities" with shape [N,] or [N, 1]
 261     opacities = torch.sigmoid(params["opacities"])
 262 
 263     dead_indices = mask.nonzero(as_tuple=True)[0]
 264     alive_indices = (~mask).nonzero(as_tuple=True)[0]
 265     n = len(dead_indices)
 266 
 267     # Sample for new GSs
 268     eps = torch.finfo(torch.float32).eps
 269     probs = opacities[alive_indices].flatten()  # ensure its shape is [N,]
 270     sampled_idxs = _multinomial_sample(probs, n, replacement=True)
 271     sampled_idxs = alive_indices[sampled_idxs]
 272     new_opacities, new_scales = compute_relocation(
 273         opacities=opacities[sampled_idxs],
 274         scales=torch.exp(params["scales"])[sampled_idxs],
 275         ratios=torch.bincount(sampled_idxs)[sampled_idxs] + 1,
 276         binoms=binoms,
 277     )
 278     new_opacities = torch.clamp(new_opacities, max=1.0 - eps, min=min_opacity)
 279 
 280     def param_fn(name: str, p: Tensor) -> Tensor:
 281         if name == "opacities":
 282             p[sampled_idxs] = torch.logit(new_opacities)
 283         elif name == "scales":
 284             p[sampled_idxs] = torch.log(new_scales)
 285         p[dead_indices] = p[sampled_idxs]
 286         return torch.nn.Parameter(p, requires_grad=p.requires_grad)
 287 
 288     def optimizer_fn(key: str, v: Tensor) -> Tensor:
 289         v[sampled_idxs] = 0
 290         return v
 291 
 292     # update the parameters and the state in the optimizers
 293     _update_param_with_optimizer(param_fn, optimizer_fn, params, optimizers)
 294     # update the extra running state
 295     for k, v in state.items():
 296         if isinstance(v, torch.Tensor):
 297             v[sampled_idxs] = 0

 343 @torch.no_grad()
 344 def inject_noise_to_position(
 345     params: Union[Dict[str, torch.nn.Parameter], torch.nn.ParameterDict],
 346     optimizers: Dict[str, torch.optim.Optimizer],
 347     state: Dict[str, Tensor],
 348     scaler: float,
 349 ):
 350     opacities = torch.sigmoid(params["opacities"].flatten())
 351     scales = torch.exp(params["scales"])
 352     covars, _ = quat_scale_to_covar_preci(
 353         params["quats"],
 354         scales,
 355         compute_covar=True,
 356         compute_preci=False,
 357         triu=False,
 358     )
 359 
 360     def op_sigmoid(x, k=100, x0=0.995):
 361         return 1 / (1 + torch.exp(-k * (x - x0)))
 362 
 363     noise = (
 364         torch.randn_like(params["means"])
 365         * (op_sigmoid(1 - opacities)).unsqueeze(-1)
 366         * scaler
 367     )
 368     noise = torch.einsum("bij,bj->bi", covars, noise)
 369     params["means"].add_(noise)

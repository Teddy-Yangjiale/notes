 247         #                这个转角误差乘上十米力臂就成了 0.13 m 的横移误差, 而且和转角误差完全相关 —— 按独立的小 sigma 喂给位姿图是错的。

 249         def solve(extra, x0, form):
 250             sh = L0 if form == "ref" else 0.0
 251             f0 = np.stack([np.sin(psi0), np.cos(psi0)], 1); pr = pri + f0 * sh
 252             Ed = dict(a=[], b=[], dx=[], dy=[], dth=[], sx=[], sy=[], sth=[])
 253             def add(*v):
 254                 for kk, vv in zip(("a", "b", "dx", "dy", "dth", "sx", "sy", "sth"), v): Ed[kk].append(vv)
 255             for k in range(K - 1):
 256                 d = pr[k + 1] - pr[k]; c, s_ = np.cos(psi0[k]), np.sin(psi0[k])
 257                 add(k, k + 1, d[0] * c - d[1] * s_, d[0] * s_ + d[1] * c, -wrap(psi0[k + 1] - psi0[k]), 0.25 if form == "u01" else 0.35, 0.6 + 0.1 * np.hypot(*d), np.radians(1.5))
 258             for r in extra:
 259                 lon_ = r[6] >= 0.06; al = r[4]
 260                 if form == "u01": add(int(r[9]), int(r[10]), r[2], r[3], -wrap(al), 0.03, 0.15 if lon_ else 2.5, np.radians(0.15))
 261                 elif form == "u01w": add(int(r[9]), int(r[10]), r[2], r[3], -wrap(al), 0.13, 0.35 if lon_ else 2.5, np.radians(0.8))
 262                 else: add(int(r[9]), int(r[10]), r[2] + L0 * np.sin(al), r[3] + L0 * np.cos(al) - L0, -wrap(al), 0.06, 0.35 if lon_ else 2.5, np.radians(0.8))
 263             if form == "vo":
 264                 # 两视几何 (k02): 相对转角准到 0.04 度, 行进方向准到 0.4 度 (= 每步横移 3 cm), 每一对都量得出; 它不给尺度, 前进量用条带的 (成峰时) 或定位间距
 265                 fwd_s = {int(r[9]): (r[3], r[6] >= 0.06) for r in extra}
 266                 Ed = dict(a=[], b=[], dx=[], dy=[], dth=[], sx=[], sy=[], sth=[])
 267                 for k in range(K - 1):
 268                     d = pr[k + 1] - pr[k]; c, s_ = np.cos(psi0[k]), np.sin(psi0[k]); lat0, fw0 = d[0] * c - d[1] * s_, d[0] * s_ + d[1] * c
 269                     fy, lon_ = fwd_s.get(k, (fw0, False))
 270                     if not lon_: fy = fw0
 271                     if VOK[k]: add(k, k + 1, fy * np.tan(VR[k, 3]), fy, -wrap(VR[k, 2]), 0.04 + 0.01 * abs(fy), 0.35 if lon_ else 0.6 + 0.1 * abs(fw0), np.radians(0.10))
 272                     else: add(k, k + 1, lat0, fy, -wrap(psi0[k + 1] - psi0[k]), 0.25, 0.35 if lon_ else 0.6 + 0.1 * abs(fw0), np.radians(1.5))
 273             Uq = dict(i=np.arange(K), E=pr[:, 0].copy(), N=pr[:, 1].copy(), psi=psi0.copy(), slat=np.full(K, 0.8), slon=np.full(K, 1.5), sth=np.full(K, np.radians(4.0)))
 274             xi = x0.copy(); xi[:, 0] += np.sin(x0[:, 2]) * sh; xi[:, 1] += np.cos(x0[:, 2]) * sh
 275             Xs = pgraph.solve(xi, {k_: np.array(v_) for k_, v_ in Ed.items()}, Uq, verbose=False)[0]
 276             Xs[:, 0] -= np.sin(Xs[:, 2]) * sh; Xs[:, 1] -= np.cos(Xs[:, 2]) * sh
 277             return Xs
 278         x0 = np.c_[pri, np.unwrap(psi0)]
 279         vf = KW + "/rot_S%g.npz" % SP; VOK = np.zeros(K - 1, bool); VR = None
 280         if os.path.exists(vf):
 281             V_ = np.load(vf)
 282             if len(V_["sta"]) == K: VR = V_["R"]; VOK = V_["ok"]
 283         X1 = solve([], x0, "u01")
 284         def run_sreg(X, win):
 285             global WIN
 286             WIN = win; jobs = []; idx = []
 287             for k in range(K - 1):
 288                 d = X[k + 1, :2] - X[k, :2]; c, s = np.cos(X[k, 2]), np.sin(X[k, 2])
 289                 jobs.append((int(sta[k]), int(sta[k + 1]), float(d[0] * c - d[1] * s), float(d[0] * s + d[1] * c), float(wrap(X[k + 1, 2] - X[k, 2])))); idx.append(k)
 290             with Pool(56) as pool: res = pool.map(sreg_one, jobs, chunksize=8)
 291             R = np.array([list(r) + [k, k + 1] for r, k in zip(res, idx) if r is not None])
 292             lat = (R[:, 5] >= 0.40) & (R[:, 7] >= 0.05) & (R[:, 8] == 0); lon = lat & (R[:, 6] >= 0.06)
 293             return R, lat, lon

 312         # ---- 轨迹对真值 ----

 346                 low = al_ * low + (1 - al_) * (al25 * low25 + (1 - al25) * low1)
 347             low = low.reshape(3, -1).T
 348             if args.robust > 0:   # 偏差用更粗尺度（默认 50 cm）的低频算：眩光是米级大块，标线在这个尺度下被平均掉
 349                 kr = int(args.robust_px * 3) * 2 + 1      # 注意别叫 k2：会盖掉畸变系数 k2
 350                 gr = torch.exp(-0.5 * ((torch.arange(kr, device=dev) - kr // 2) / args.robust_px) ** 2); gr = gr / gr.sum()
 351                 def blur2(x):
 352                     x = Fnn.conv2d(x[:, None], gr.view(1, 1, 1, -1), padding=(0, kr // 2))
 353                     return Fnn.conv2d(x, gr.view(1, 1, -1, 1), padding=(kr // 2, 0))[:, 0]
 354                 low2 = (blur2(fm) / blur2(mk.reshape(1, TPI, TPI)).clamp_min(1e-3)).reshape(3, -1).T
 355             hi = col - low[idx]
 356             # 低频（颜色/亮度）：影子区几乎不参与、后摄减半——影子区和后摄只贡献高频纹理
 357             wl = f_ * (3.0 / dist_) ** args.lowpow * (1 - (1 - args.low_shadow) * S.clamp(0, 1)) * (args.low_rear if ch == 1 else 1.0)
 358             accL.index_add_(0, idx[ok], low[idx][ok] * wl[ok, None]); wsL.index_add_(0, idx[ok], wl[ok])
 359             acc.index_add_(0, idx[ok], hi[ok] * w[ok, None]); ws.index_add_(0, idx[ok], w[ok])
 360             if args.seam:
 361                 seam_store.append((idx[ok], hi[ok].half(), w[ok], S[ok].float(), low[idx][ok].half())); seam_vi.append(vi)
 362             if args.low_adapt:
 363                 lum_ = (low[idx][ok] * LUMW).sum(1)
 364                 accM1.index_add_(0, idx[ok], lum_ * wl[ok]); accM2.index_add_(0, idx[ok], lum_ ** 2 * wl[ok])
 365             if args.robust > 0:
 366                 store.append((idx[ok], low[idx][ok], hi[ok], wl[ok], w[ok], low2[idx][ok]))
 367         else:
 368             acc.index_add_(0, idx[ok], col[ok] * w[ok, None]); ws.index_add_(0, idx[ok], w[ok])
 369     if os.environ.get("P14_DEBUG"):
 370         print("  debug pass1: store", len(store), "cov", int((ws > 1e-7).sum()), "nan ws", int(torch.isnan(ws).sum()), "nan l2", sum(int(torch.isnan(x[5]).sum()) for x in store) if store else 0, flush=True)
 371     if args.robust > 0 and store:
 372         # 第二遍：共识低频 → 每张的 RGB 偏差 → 柯西权重，低频高频一起重新累加
 373         # 眩光只会加亮、加彩色：比共识亮的偏差罚得重（σ），暗的偏差罚得轻（3σ），色度偏差按 σ；迭代 3 次逼近众数

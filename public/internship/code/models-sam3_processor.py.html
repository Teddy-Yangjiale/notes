  41     @torch.inference_mode()
  42     def set_image(self, image, state=None):
  43         """Sets the image on which we want to do predictions."""
  44         if state is None:
  45             state = {}
  46 
  47         if isinstance(image, PIL.Image.Image):
  48             width, height = image.size
  49         elif isinstance(image, (torch.Tensor, np.ndarray)):
  50             height, width = image.shape[-2:]
  51         else:
  52             raise ValueError("Image must be a PIL image or a tensor")
  53 
  54         image = v2.functional.to_image(image).to(self.device)
  55         image = self.transform(image).unsqueeze(0)
  56 
  57         state["original_height"] = height
  58         state["original_width"] = width
  59         state["backbone_out"] = self.model.backbone.forward_image(image)
  60         inst_interactivity_en = self.model.inst_interactive_predictor is not None
  61         if inst_interactivity_en and "sam2_backbone_out" in state["backbone_out"]:
  62             sam2_backbone_out = state["backbone_out"]["sam2_backbone_out"]
  63             sam2_backbone_out["backbone_fpn"][0] = (
  64                 self.model.inst_interactive_predictor.model.sam_mask_decoder.conv_s0(
  65                     sam2_backbone_out["backbone_fpn"][0]
  66                 )
  67             )
  68             sam2_backbone_out["backbone_fpn"][1] = (
  69                 self.model.inst_interactive_predictor.model.sam_mask_decoder.conv_s1(
  70                     sam2_backbone_out["backbone_fpn"][1]
  71                 )
  72             )
  73         return state

 113     @torch.inference_mode()
 114     def set_text_prompt(self, prompt: str, state: Dict):
 115         """Sets the text prompt and run the inference"""
 116 
 117         if "backbone_out" not in state:
 118             raise ValueError("You must call set_image before set_text_prompt")
 119 
 120         text_outputs = self.model.backbone.forward_text([prompt], device=self.device)
 121         # will erase the previous text prompt if any
 122         state["backbone_out"].update(text_outputs)
 123         if "geometric_prompt" not in state:
 124             state["geometric_prompt"] = self.model._get_dummy_prompt()
 125 
 126         return self._forward_grounding(state)
 127 
 128     @torch.inference_mode()
 129     def add_geometric_prompt(self, box: List, label: bool, state: Dict):
 130         """Adds a box prompt and run the inference.
 131         The image needs to be set, but not necessarily the text prompt.
 132         The box is assumed to be in [center_x, center_y, width, height] format and normalized in [0, 1] range.
 133         The label is True for a positive box, False for a negative box.
 134         """
 135         if "backbone_out" not in state:
 136             raise ValueError("You must call set_image before set_text_prompt")
 137 
 138         if "language_features" not in state["backbone_out"]:
 139             # Looks like we don't have a text prompt yet. This is allowed, but we need to set the text prompt to "visual" for the model to rely only on the geometric prompt
 140             dummy_text_outputs = self.model.backbone.forward_text(
 141                 ["visual"], device=self.device
 142             )
 143             state["backbone_out"].update(dummy_text_outputs)
 144 
 145         if "geometric_prompt" not in state:
 146             state["geometric_prompt"] = self.model._get_dummy_prompt()
 147 
 148         # adding a batch and sequence dimension
 149         boxes = torch.tensor(box, device=self.device, dtype=torch.float32).view(1, 1, 4)
 150         labels = torch.tensor([label], device=self.device, dtype=torch.bool).view(1, 1)
 151         state["geometric_prompt"].append_boxes(boxes, labels)
 152 
 153         return self._forward_grounding(state)
 154 
 155     def reset_all_prompts(self, state: Dict):
 156         """Removes all the prompts and results"""
 157         if "backbone_out" in state:
 158             backbone_keys_to_del = [
 159                 "language_features",
 160                 "language_mask",
 161                 "language_embeds",
 162             ]
 163             for key in backbone_keys_to_del:
 164                 if key in state["backbone_out"]:
 165                     del state["backbone_out"][key]
 166 
 167         keys_to_del = ["geometric_prompt", "boxes", "masks", "masks_logits", "scores"]
 168         for key in keys_to_del:
 169             if key in state:
 170                 del state[key]

 172     @torch.inference_mode()
 173     def set_confidence_threshold(self, threshold: float, state=None):
 174         """Sets the confidence threshold for the masks"""
 175         self.confidence_threshold = threshold
 176         if state is not None and "boxes" in state:
 177             # we need to filter the boxes again
 178             # In principle we could do this more efficiently since we would only need
 179             # to rerun the heads. But this is simpler and not too inefficient
 180             return self._forward_grounding(state)
 181         return state

 183     @torch.inference_mode()
 184     def _forward_grounding(self, state: Dict):
 185         outputs = self.model.forward_grounding(
 186             backbone_out=state["backbone_out"],
 187             find_input=self.find_stage,
 188             geometric_prompt=state["geometric_prompt"],
 189             find_target=None,
 190         )
 191 
 192         out_bbox = outputs["pred_boxes"]
 193         out_logits = outputs["pred_logits"]
 194         out_masks = outputs["pred_masks"]
 195         out_probs = out_logits.sigmoid()
 196         presence_score = outputs["presence_logit_dec"].sigmoid().unsqueeze(1)
 197         out_probs = (out_probs * presence_score).squeeze(-1)
 198 
 199         keep = out_probs > self.confidence_threshold
 200         out_probs = out_probs[keep]
 201         out_masks = out_masks[keep]
 202         out_bbox = out_bbox[keep]
 203 
 204         # convert to [x0, y0, x1, y1] format
 205         boxes = box_ops.box_cxcywh_to_xyxy(out_bbox)
 206 
 207         img_h = state["original_height"]
 208         img_w = state["original_width"]
 209         scale_fct = torch.tensor([img_w, img_h, img_w, img_h]).to(self.device)
 210         boxes = boxes * scale_fct[None, :]
 211 
 212         out_masks = interpolate(
 213             out_masks.unsqueeze(1),
 214             (img_h, img_w),
 215             mode="bilinear",
 216             align_corners=False,
 217         ).sigmoid()
 218 
 219         state["masks_logits"] = out_masks
 220         state["masks"] = out_masks > 0.5
 221         state["boxes"] = boxes
 222         state["scores"] = out_probs
 223         return state

  22 def _load_raw_pi3_checkpoint(ckpt_path: str):
  23     """Load pickled / safetensors weights. Supports local paths and ``oss://bucket/...`` via ``ossutil.oss``.
  24 
  25     ``OssUtil.open`` buffers the entire object in memory before deserialize; treat large checkpoints like full-RAM downloads.
  26     """
  27     ckpt_path = str(ckpt_path)
  28     use_oss = ckpt_path.startswith("oss://")
  29     if ckpt_path.lower().endswith(".safetensors"):
  30         if use_oss:
  31             from datasets.vendored_dust3r.utils.ossutil import oss_ori
  32 
  33             return safetensors_load_bytes(oss_ori.open(ckpt_path).read())
  34         return load_file(ckpt_path, device="cpu")
  35 
  36     # .bin / .pt / .pth etc.
  37     if use_oss:
  38         from datasets.vendored_dust3r.utils.ossutil import oss_ori
  39 
  40         bio = oss_ori.open(ckpt_path)
  41         return torch.load(bio, weights_only=False, map_location="cpu")
  42 
  43     return torch.load(ckpt_path, weights_only=False, map_location="cpu")

  56 def _resolve_confidence_training_flags(
  57     *,
  58     train_conf: bool,
  59     enable_confidence: Optional[bool],
  60     confidence_only_train: Optional[bool],
  61 ) -> Tuple[bool, bool]:
  62     """Resolve new confidence controls while preserving legacy ``train_conf`` semantics.
  63 
  64     Historically, ``train_conf=True`` both created the confidence branch and froze
  65     every prediction branch except confidence.  New training recipes need to enable
  66     confidence during joint full fine-tuning, so construction and freezing are now
  67     separate opt-in controls.  Leaving both new arguments unset is exactly backward
  68     compatible.
  69     """
  70 
  71     legacy_train_conf = bool(train_conf)
  72     enabled = legacy_train_conf if enable_confidence is None else bool(enable_confidence)
  73     confidence_only = (
  74         legacy_train_conf
  75         if confidence_only_train is None
  76         else bool(confidence_only_train)
  77     )
  78     if confidence_only and not enabled:
  79         raise ValueError(
  80             "confidence_only_train=True requires enable_confidence=True "
  81             "(or legacy train_conf=True)."
  82         )
  83     return enabled, confidence_only

 383     def _initialize_conf_decoder_from_point(self) -> None:
 384         """Warm-start a missing confidence decoder from the trained point decoder."""
 385 
 386         res = self.conf_decoder.load_state_dict(self.point_decoder.state_dict(), strict=True)
 387         if res.missing_keys or res.unexpected_keys:
 388             raise RuntimeError(
 389                 "conf_decoder and point_decoder are expected to be isomorphic; "
 390                 f"got {res}"
 391             )
 392         print(
 393             "[Pi3] Checkpoint has no conf_decoder weights; initialized conf_decoder "
 394             "from the loaded point_decoder.",
 395             flush=True,
 396         )

 398     def _freeze_for_confidence_only(self) -> Tuple[str, ...]:
 399         """Freeze the complete model except ``conf_decoder`` and ``conf_head``."""
 400 
 401         prefixes = ("conf_decoder.", "conf_head.")
 402         trainable = []
 403         for name, param in self.named_parameters():
 404             should_train = name.startswith(prefixes)
 405             param.requires_grad_(should_train)
 406             if should_train:
 407                 trainable.append(name)
 408         if not trainable:
 409             raise RuntimeError(
 410                 "Confidence-only training found no conf_decoder/conf_head parameters."
 411             )
 412         leaked = [
 413             name
 414             for name, param in self.named_parameters()
 415             if param.requires_grad and not name.startswith(prefixes)
 416         ]
 417         if leaked:
 418             raise RuntimeError(
 419                 "Confidence-only parameter freeze leaked non-confidence parameters: "
 420                 + ", ".join(leaked[:16])
 421             )
 422         print(
 423             "[Pi3] Confidence-only training: trainable parameters are restricted to "
 424             f"conf_decoder/conf_head ({sum(self.get_parameter(n).numel() for n in trainable) / 1e6:.3f}M).",
 425             flush=True,
 426         )
 427         return tuple(trainable)

 433     def decode(
 434         self,
 435         hidden,
 436         N,
 437         H,
 438         W,
 439         causal_global: bool = True,
 440         stream_use_cache: bool = False,
 441         past_key_values: Optional[List[Optional[Tuple[torch.Tensor, torch.Tensor]]]] = None,
 442     ) -> Tuple[torch.Tensor, torch.Tensor, Optional[List[Optional[Tuple[torch.Tensor, torch.Tensor]]]]]:
 443         BN, hw, _ = hidden.shape
 444         B = BN // N
 445 
 446         final_output = []
 447 
 448         hidden = hidden.reshape(B * N, hw, -1)
 449 
 450         register_token = self.register_token.repeat(B, N, 1, 1).reshape(
 451             B * N, *self.register_token.shape[-2:]
 452         )
 453 
 454         hidden = torch.cat([register_token, hidden], dim=1)
 455         hw = hidden.shape[1]
 456 
 457         if self.pos_type.startswith("rope"):
 458             pos = self.position_getter(
 459                 B * N, H // self.patch_size, W // self.patch_size, hidden.device
 460             )
 461 
 462         if self.patch_start_idx > 0:
 463             pos = pos + 1
 464             pos_special = torch.zeros(B * N, self.patch_start_idx, 2).to(hidden.device).to(pos.dtype)
 465             pos = torch.cat([pos_special, pos], dim=1)
 466 
 467         if stream_use_cache and past_key_values is None:
 468             past_key_values = [None] * self.num_global_decoder_blocks
 469 
 470         global_blk_idx = 0
 471 
 472         for i in range(len(self.decoder)):
 473             blk = self.decoder[i]
 474 
 475             if i % 2 == 0:
 476                 pos = pos.reshape(B * N, hw, -1)
 477                 hidden = hidden.reshape(B * N, hw, -1)
 478             else:
 479                 pos = pos.reshape(B, N * hw, -1)
 480                 hidden = hidden.reshape(B, N * hw, -1)
 481 
 482             attn_mask = None
 483             if i % 2 == 1 and causal_global and not stream_use_cache:
 484                 attn_mask = causal_frame_attention_mask(
 485                     N, hw, hidden.dtype, hidden.device
 486                 )
 487 
 488             use_blk_cache = stream_use_cache and (i % 2 == 1)
 489             past_kv = (
 490                 past_key_values[global_blk_idx]
 491                 if use_blk_cache and past_key_values is not None
 492                 else None
 493             )
 494 
 495             if use_blk_cache:
 496                 hidden, new_kv = blk(
 497                     hidden,
 498                     xpos=pos,
 499                     attn_mask=None,
 500                     past_key_values=past_kv,
 501                     use_cache=True,
 502                 )
 503                 assert past_key_values is not None
 504                 past_key_values[global_blk_idx] = new_kv
 505             elif (
 506                 i >= self.num_dec_blk_not_to_checkpoint
 507                 and self.training
 508                 and not stream_use_cache
 509             ):
 510                 # Same as Pi3_train: `checkpoint(blk, hidden, xpos=pos, ...)`.
 511                 # Do NOT wrap in `lambda ... xpos=pos ...`: `pos` is reassigned every iteration;
 512                 # Python late-binds `pos` so backward recompute sees the wrong shape (725 vs N*hw).
 513                 hidden = checkpoint(
 514                     blk,
 515                     hidden,
 516                     xpos=pos,
 517                     attn_mask=attn_mask,
 518                     past_key_values=None,
 519                     use_cache=False,
 520                     use_reentrant=False,
 521                 )
 522             else:
 523                 hidden = blk(
 524                     hidden,
 525                     xpos=pos,
 526                     attn_mask=attn_mask,
 527                     past_key_values=None,
 528                     use_cache=False,
 529                 )
 530 
 531             if i % 2 == 1:
 532                 global_blk_idx += 1
 533 
 534             if i + 1 in [len(self.decoder) - 1, len(self.decoder)]:
 535                 final_output.append(hidden.reshape(B * N, hw, -1))
 536 
 537         fused = torch.cat([final_output[0], final_output[1]], dim=-1)
 538         pos_out = pos.reshape(B * N, hw, -1)
 539         return fused, pos_out, past_key_values

 541     def _predict_camera_poses(
 542         self,
 543         camera_tokens: torch.Tensor,
 544         *,
 545         B: int,
 546         N: int,
 547         patch_h: int,
 548         patch_w: int,
 549         camera_state: Optional[Dict[str, torch.Tensor]] = None,
 550         return_state: bool = False,
 551     ):
 552         if self.camera_pose_mode == "relative_adjacent":
 553             feat = camera_tokens.reshape(B, N, -1, camera_tokens.shape[-1])
 554             return self.camera_head(
 555                 feat,
 556                 camera_state=camera_state,
 557                 return_state=return_state,
 558             )
 559         poses = self.camera_head(camera_tokens, patch_h, patch_w).reshape(B, N, 4, 4)
 560         if return_state:
 561             return poses, None
 562         return poses

 564     def forward(
 565         self,
 566         imgs: torch.Tensor,
 567         causal_global_attn: Optional[bool] = None,
 568         stream_use_cache: bool = False,
 569         past_key_values: Optional[List[Optional[Tuple[torch.Tensor, torch.Tensor]]]] = None,
 570         ref_hidden: Optional[torch.Tensor] = None,
 571         camera_state: Optional[Dict[str, torch.Tensor]] = None,
 572     ) -> Dict[str, Any]:
 573         if causal_global_attn is None:
 574             causal_global_attn = self.causal_global_attn
 575 
 576         imgs = (imgs - self.image_mean) / self.image_std
 577 
 578         B, N, _, H, W = imgs.shape
 579         if stream_use_cache and N != 1:
 580             raise ValueError("stream_use_cache requires N==1 (single-frame steps with KV cache)")
 581         patch_h, patch_w = H // 14, W // 14
 582         tks = patch_h * patch_w + self.patch_start_idx
 583 
 584         imgs_bn = imgs.reshape(B * N, _, H, W)
 585         hidden = self.encoder(imgs_bn, is_training=True)
 586 
 587         if isinstance(hidden, dict):
 588             hidden = hidden["x_norm_patchtokens"]
 589 
 590         hidden, pos, past_key_values = self.decode(
 591             hidden,
 592             N,
 593             H,
 594             W,
 595             causal_global=causal_global_attn,
 596             stream_use_cache=stream_use_cache,
 597             past_key_values=past_key_values,
 598         )
 599 
 600         new_ref_hidden: Optional[torch.Tensor] = None
 601         if self.use_global_points:
 602             if stream_use_cache and N == 1:
 603                 hf = hidden.reshape(B, tks, -1)
 604                 if ref_hidden is None:
 605                     context = hf
 606                     new_ref_hidden = hf.detach()
 607                 else:
 608                     context = ref_hidden
 609                     new_ref_hidden = ref_hidden
 610             else:
 611                 context = (
 612                     hidden.reshape(B, N, tks, -1)[:, 0:1]
 613                     .repeat(1, N, 1, 1)
 614                     .reshape(B * N, tks, -1)
 615                 )
 616             global_point_hidden = self.global_points_decoder(
 617                 hidden, context, xpos=pos, ypos=pos
 618             )
 619 
 620         point_hidden = self.point_decoder(hidden, xpos=pos)
 621         if self.train_conf:
 622             conf_hidden = self.conf_decoder(hidden, xpos=pos)
 623         camera_hidden = self.camera_decoder(hidden, xpos=pos)
 624 
 625         with torch.amp.autocast(device_type="cuda", enabled=False):
 626             point_hidden = point_hidden.float()
 627             ret = self.point_head(
 628                 [point_hidden[:, self.patch_start_idx :]], (H, W)
 629             ).reshape(B, N, H, W, -1)
 630             xy, z = ret.split([2, 1], dim=-1)
 631             z = depth_from_log_z(z, self.point_z_log_max)
 632             local_points = torch.cat([xy * z, z], dim=-1)
 633 
 634             if self.train_conf:
 635                 conf_hidden = conf_hidden.float()
 636                 conf = self.conf_head(
 637                     [conf_hidden[:, self.patch_start_idx :]], (H, W)
 638                 ).reshape(B, N, H, W, -1)
 639             else:
 640                 conf = None
 641 
 642             camera_hidden = camera_hidden.float()
 643             camera_tokens_for_head = camera_hidden[:, self.patch_start_idx :]
 644             if (
 645                 self.camera_pose_mode == "relative_adjacent"
 646                 and self.relative_camera_head_type == "token_pair"
 647             ):
 648                 camera_tokens_for_head = camera_hidden
 649             camera_poses, new_camera_state = self._predict_camera_poses(
 650                 camera_tokens_for_head,
 651                 B=B,
 652                 N=N,
 653                 patch_h=patch_h,
 654                 patch_w=patch_w,
 655                 camera_state=camera_state,
 656                 return_state=True,
 657             )
 658 
 659             if self.use_global_points:
 660                 global_point_hidden = global_point_hidden.float()
 661                 global_points = self.global_point_head(
 662                     [global_point_hidden[:, self.patch_start_idx :]], (H, W)
 663                 ).reshape(B, N, H, W, -1)
 664             else:
 665                 global_points = None
 666 
 667             points = torch.einsum(
 668                 "bnij, bnhwj -> bnhwi",
 669                 camera_poses,
 670                 homogenize_points(local_points),
 671             )[..., :3]
 672 
 673         out: Dict[str, Any] = dict(
 674             points=points,
 675             local_points=local_points,
 676             conf=conf,
 677             camera_poses=camera_poses,
 678             global_points=global_points,
 679         )
 680         if stream_use_cache:
 681             out["past_key_values"] = past_key_values
 682         if new_ref_hidden is not None:
 683             out["ref_hidden"] = new_ref_hidden
 684         if new_camera_state is not None:
 685             out["camera_state"] = new_camera_state
 686         return out
 687 
 688     def inference_stream(
 689         self,
 690         frames: List[torch.Tensor],
 691         return_stacked: bool = False,
 692     ) -> Union[List[Dict[str, Any]], Dict[str, Any]]:
 693         past_key_values: List[Optional[Tuple[torch.Tensor, torch.Tensor]]] = [
 694             None
 695         ] * self.num_global_decoder_blocks
 696         ref_hidden: Optional[torch.Tensor] = None
 697         camera_state: Optional[Dict[str, torch.Tensor]] = None
 698         outputs: List[Dict[str, Any]] = []
 699         for fr in frames:
 700             x = fr
 701             if x.dim() == 3:
 702                 x = x.unsqueeze(0)
 703             x = x.unsqueeze(1)
 704             pred = self.forward(
 705                 x,
 706                 causal_global_attn=True,
 707                 stream_use_cache=True,
 708                 past_key_values=past_key_values,
 709                 ref_hidden=ref_hidden,
 710                 camera_state=camera_state,
 711             )
 712             past_key_values = pred["past_key_values"]
 713             ref_hidden = pred.get("ref_hidden")
 714             camera_state = pred.get("camera_state", camera_state)
 715             clean = {
 716                 k: v
 717                 for k, v in pred.items()
 718                 if k not in ("past_key_values", "ref_hidden", "camera_state")
 719             }
 720             outputs.append(clean)
 721         if not return_stacked:
 722             return outputs
 723         stacked: Dict[str, Any] = {}
 724         for k in outputs[0].keys():
 725             vals = [o[k] for o in outputs]
 726             if vals[0] is None:
 727                 stacked[k] = None
 728             else:
 729                 stacked[k] = torch.cat(vals, dim=1)
 730         return stacked

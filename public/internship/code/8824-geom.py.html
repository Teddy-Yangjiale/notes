  57 def triangulate(origins, dirs, weights=None):
  58     """最小二乘多视三角化：min Σ wᵢ‖(I − dᵢdᵢᵀ)(X − oᵢ)‖²。
  59 
  60     返回 (X, 残差RMS, 条件数)。条件数大 = 几何退化（前向运动的极点附近）。
  61     """
  62     O = np.asarray(origins, float)
  63     D = np.asarray(dirs, float)
  64     w = np.ones(len(O)) if weights is None else np.asarray(weights, float)
  65     A = np.zeros((3, 3)); b = np.zeros(3)
  66     for oi, di, wi in zip(O, D, w):
  67         P = np.eye(3) - np.outer(di, di)
  68         A += wi * P
  69         b += wi * (P @ oi)
  70     cond = float(np.linalg.cond(A))
  71     if not np.isfinite(cond) or cond > 1e12:
  72         return None, np.inf, cond
  73     X = np.linalg.solve(A, b)
  74     res = np.array([np.linalg.norm((np.eye(3) - np.outer(d, d)) @ (X - o))
  75                     for o, d in zip(O, D)])
  76     return X, float(np.sqrt((res ** 2).mean())), cond

  97 def robust_triangulate(obs, K, huber_px=8.0, max_iter=8):
  98     """带 Huber 重加权的三角化。obs = [(R_cw, C, u, v)]。
  99 
 100     返回 dict：X / rms_px / cond / max_angle / n_used / 每次观测的重投影残差。
 101     """
 102     K_inv = np.linalg.inv(K)
 103     O, D = [], []
 104     for R_cw, C, u, v in obs:
 105         o, d = ray(K_inv, R_cw, C, u, v)
 106         O.append(o); D.append(d)
 107     w = np.ones(len(obs))
 108     X = None
 109     for _ in range(max_iter):
 110         X, _, cond = triangulate(O, D, w)
 111         if X is None:
 112             return None
 113         err = []
 114         for (R_cw, C, u, v) in obs:
 115             pr = reproject(X, K, R_cw, C)
 116             err.append(np.hypot(pr[0] - u, pr[1] - v) if pr else 1e6)
 117         err = np.array(err)
 118         nw = np.where(err <= huber_px, 1.0, huber_px / np.maximum(err, 1e-6))
 119         if np.allclose(nw, w, atol=1e-3):
 120             w = nw; break
 121         w = nw
 122     used = w > 0.3
 123     return {"X": X, "err_px": err, "w": w, "cond": cond,
 124             "rms_px": float(np.sqrt((err[used] ** 2).mean())) if used.any() else float("inf"),
 125             "max_angle": max_pair_angle(np.asarray(D)[used]) if used.sum() >= 2 else 0.0,
 126             "n_used": int(used.sum())}

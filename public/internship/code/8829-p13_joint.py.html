 250 def proj_np(Xc):
 251     """相机系点 → 像素（OPENCV 畸变，与 torch 版 project 一致；不依赖 cv2.projectPoints 的版本差异）"""
 252     x = Xc[:, 0] / Xc[:, 2]; y = Xc[:, 1] / Xc[:, 2]; r2 = x * x + y * y; rad = 1 + k1 * r2 + k2 * r2 * r2
 253     xd = x * rad + 2 * p1 * x * y + p2 * (r2 + 2 * x * x); yd = y * rad + p1 * (r2 + 2 * y * y) + 2 * p2 * x * y
 254     return np.c_[fx * xd + cx, fy * yd + cy]

 337 def so3(w):
 338     th = torch.linalg.norm(w) + 1e-12; kx = w / th
 339     Kx = torch.zeros((3, 3), dtype=w.dtype, device=w.device)
 340     Kx[0, 1], Kx[0, 2], Kx[1, 0], Kx[1, 2], Kx[2, 0], Kx[2, 1] = -kx[2], kx[1], kx[2], -kx[0], -kx[1], kx[0]
 341     return torch.eye(3, dtype=w.dtype, device=w.device) + torch.sin(th) * Kx + (1 - torch.cos(th)) * Kx @ Kx

 347 def so3_b(w):   # 批量 Rodrigues：N×3 → N×3×3
 348     th = torch.linalg.norm(w, dim=1, keepdim=True) + 1e-12; k = w / th
 349     Kx = torch.zeros((len(w), 3, 3), dtype=w.dtype, device=w.device)
 350     Kx[:, 0, 1], Kx[:, 0, 2], Kx[:, 1, 0], Kx[:, 1, 2], Kx[:, 2, 0], Kx[:, 2, 1] = -k[:, 2], k[:, 1], k[:, 2], -k[:, 0], -k[:, 1], k[:, 0]
 351     st = torch.sin(th)[:, :, None]; ct = torch.cos(th)[:, :, None]
 352     return torch.eye(3, dtype=w.dtype, device=w.device)[None] + st * Kx + (1 - ct) * Kx @ Kx

 360 def sv_terms():
 361     """每站：车体原点离路面高差、车头方向与路面法向的点积（≈ 相对路面的俯仰 rad）、车右方向与法向的点积（≈ 横滚）"""

 369     gN = (zs_at(p0[:, 0], p0[:, 1] + h) - zs_at(p0[:, 0], p0[:, 1] - h)) / (2 * h)
 370     ns = torch.stack([-gE, -gN, torch.ones_like(gE)], -1); ns = ns / torch.linalg.norm(ns, dim=1, keepdim=True)
 371     return p0[:, 2] - zc, (fw * ns).sum(1), (rt * ns).sum(1)

 380 def project(vi, ii):
 381     r, ch, fi = views[vi]
 382     S_ = so3(omega[fi]); Rf = S_ @ W0_t[fi, :3, :3]
 383     tf = (W0_t[fi, :3, 3] if args.rot_world else S_ @ W0_t[fi, :3, 3]) + delta[fi]   # 绕相机中心：t' = S·t，相机位置不随转角动
 384     X = torch.stack([tE[ii], tN[ii], heights(ii)], -1)
 385     if ch == 2:
 386         R, t = Rf, tf; Xw = X
 387     else:
 388         R = R_rf_t @ Rf; t = R_rf_t @ tf + t_rf_t
 389         if args.rear_ext:
 390             Rx_ = so3(rext_w); R = Rx_ @ R; t = Rx_ @ t + rext_t
 391         d = rinit[vi] + rear_d[vi]
 392         # 后摄修正：绕相机中心的偏航 + 水平平移（作用在世界点上）
 393         Cc = -R.T @ t
 394         cyw, syw = torch.cos(d[2]), torch.sin(d[2])
 395         Xw = X.clone()
 396         x0 = X[:, 0] - d[0] - Cc[0]; y0 = X[:, 1] - d[1] - Cc[1]
 397         Xw = torch.stack([cyw * x0 + syw * y0 + Cc[0], -syw * x0 + cyw * y0 + Cc[1], X[:, 2]], -1)
 398     Xc = Xw @ R.T + t
 399     if args.rolling:
 400         # 先按名义位姿求像素行 → 该行相对画面中心行的拍摄时刻差 τ → 世界点相对相机平移 −v·τ，再投一次
 401         yr = (Xc[:, 1] / Xc[:, 2]) * fy + cy
 402         tau = (yr - IMG_H / 2) / IMG_H * rs[0 if ch == 2 else 1]
 403         Xc = (Xw - tau[:, None] * VEL_t[fi][None]) @ R.T + t
 404     x = Xc[:, 0] / Xc[:, 2]; y = Xc[:, 1] / Xc[:, 2]
 405     r2 = x * x + y * y; rad = 1 + k1 * r2 + k2 * r2 * r2
 406     xd = x * rad + 2 * p1 * x * y + p2 * (r2 + 2 * x * x); yd = y * rad + p1 * (r2 + 2 * y * y) + 2 * p2 * x * y
 407     u = fx * xd + cx; v = fy * yd + cy
 408     u = u.float(); v = v.float()
 409     if args.nonrigid:
 410         # 画面上（路面区：行 OSD..BOT）的双线性形变：控制点覆盖整幅宽、行从 0.3H 到 BOT
 411         gu = (u / IMG_W) * 2 - 1; gv = ((v - 0.3 * IMG_H) / (BOTV[ch] - 0.3 * IMG_H)) * 2 - 1
 412         d = Fnn.grid_sample(nrg[vi][None], torch.stack([gu, gv], -1)[None, :, None, :], mode="bilinear", padding_mode="border", align_corners=True)[0, :, :, 0]
 413         u = u + d[0]; v = v + d[1]
 414     return u, v

 417 def observe(vi, ii):
 418     """返回该照片对这些纹理点给出的'纹理估计' pred_T（去掉曝光/暗角/影子）、原始取样、有效"""
 419     r, ch, fi = views[vi]
 420     u, v = project(vi, ii)
 421     g = torch.stack([(u + 0.5) / IMG_W * 2 - 1, (v + 0.5) / IMG_H * 2 - 1], -1)[None, :, None, :]
 422     I = Fnn.grid_sample(imgs[vi][None].float(), g, mode="bilinear", align_corners=False)[0, :, :, 0].T     # N×3
 423     V = torch.exp(Fnn.grid_sample(vig[0 if ch == 2 else 1][None], g, mode="bilinear", align_corners=True)[0, 0, :, 0])
 424     S = Fnn.grid_sample(shadows[vi][None].float(), g, mode="bilinear", align_corners=False)[0, 0, :, 0]
 425     gain = torch.exp(loga[vi])[None] * V[:, None] * torch.exp(shad[vi][None] * S[:, None])
 426     add = bb[vi][None]
 427     if args.glare_grid:
 428         add = add + Fnn.grid_sample(bgrid[vi][None], g, mode="bilinear", padding_mode="border", align_corners=True)[0, :, :, 0].T
 429     pred = (I - 10.0 * add) / gain       # bb 以 10 个灰度为单位
 430     valid = (u > 1) & (u < IMG_W - 2) & (v > OSD_ROWS + 3) & (v < IMG_H - 2)
 431     return pred, valid
